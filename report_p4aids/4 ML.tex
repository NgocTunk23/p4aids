\section{Triển khai thuật toán Học Máy - Machine Learning}

\subsection{Thước đo Đánh giá và Cơ chế Ngưỡng Quyết định}

\subsubsection{Thước đo Đánh giá}
Trong bài toán phát hiện dị thường với tỷ lệ mất cân bằng nghiêm trọng ($97{,}87\%$ bình thường vs $2{,}13\%$ dị thường), việc lựa chọn thước đo đánh giá là yếu tố sống còn để đảm bảo tính khách quan:
\begin{itemize}
    \item \textbf{Vô hiệu hóa Độ chính xác Tổng thể (Accuracy):} Một mô hình ngây thơ dự đoán $100\%$ mẫu là bình thường ($y=0$) vẫn đạt $\text{Accuracy} = 97{,}87\%$, nhưng hoàn toàn thất bại trong việc cảnh báo sự cố rò rỉ hay quá tải điện.
    \item \textbf{Sự sai lệch của ROC-AUC:} Do số lượng mẫu âm tính quá lớn ($TN > 1{,}71$ triệu dòng), tỷ lệ báo động giả $\text{FPR} = \frac{FP}{FP+TN}$ luôn bị ghìm ở mức cực nhỏ dù mô hình dự đoán nhầm hàng chục ngàn mẫu. Điều này tạo ra ảo tưởng lạc quan với điểm $\text{ROC-AUC} > 0{,}95$ nhưng thực chất độ chính xác dự báo dị thường lại rất thấp.
    \item \textbf{F1-Score là Thước đo Đánh giá Tối thượng (Primary Metric):} Mục tiêu vận hành thực tế của hệ thống điện đòi hỏi: vừa phải bắt trúng tối đa các sự cố bất thường để kịp thời xử lý (tối đa hóa Recall), vừa phải giảm thiểu các báo động giả gây lãng phí nhân lực kiểm tra hiện trường (tối đa hóa Precision). $F_1$-Score chính là trung bình điều hòa (\textit{harmonic mean}) ràng buộc chặt chẽ cả hai đại lượng:
    \begin{equation*}
        \text{Precision} = \frac{TP}{TP + FP}, \quad \text{Recall} = \frac{TP}{TP + FN}
    \end{equation*}
    \begin{equation*}
        F_1\text{-Score} = 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}
    \end{equation*}
    Nếu một trong hai chỉ số Precision hoặc Recall bị suy giảm, $F_1$-Score sẽ lập tức bị kéo tụt. Toàn bộ các vòng lặp tối ưu hóa và so sánh mô hình trong chương này đều sử dụng $F_1$-Score làm hàm mục tiêu quyết định.
\end{itemize}

\subsubsection{Cơ chế Ngưỡng Quyết định}\hspace{1cm}
Do nhãn $1$ quá khan hiếm, hàm mất mát Log-loss đẩy xác suất dự đoán $P(y=1)$ lệch mạnh về gần $0$. Nhóm thiết lập cơ chế ra quyết định hai tầng:
\begin{enumerate}
    \item \textbf{Tầng 1 (Dự báo Xác suất Liên tục):} Mô hình tính toán xác suất dị thường liên tục $\hat{p}_i = P(y_i=1 \mid \mathbf{x}_i) \in [0, 1]$.
    \item \textbf{Tầng 2 (Quyết định Gán nhãn qua Ngưỡng Tối ưu):} Nhãn nhị phân được gán thông qua ngưỡng cắt $\tau$:
    \begin{equation*}
        \hat{y}_i = \mathbb{1}\left[\hat{p}_i \ge \tau\right]
    \end{equation*}
    trong đó $\tau$ không cố định ở mức $0{,}5$ mà được quét lưới để cực đại hóa $F_1$-Score trên tập kiểm định:
    \begin{equation*}
        \tau^* = \arg\max_{\tau \in [0{,}01; 0{,}99]} F_1\left(y_{\text{val}}, \, \mathbb{1}[\hat{p}_{\text{val}} \geq \tau]\right)
    \end{equation*}
\end{enumerate}

\subsubsection{Tinh chỉnh Siêu tham số (Hyperparameter Tuning) với Optuna}

Nhằm khai thác tối đa tiềm năng của từng thuật toán, nhóm sử dụng khung tối ưu hóa \textbf{Optuna} dựa trên thuật toán Bayes TPE (\textit{Tree-structured Parzen Estimator}). Không gian tìm kiếm được cấu hình riêng cho từng thuật toán qua các chu kỳ thử nghiệm (\textit{trials}).

Do dữ liệu có mức độ mất cân bằng lớp cao và thang đo đánh giá cốt lõi của bài toán là chỉ số $F_1$, hàm mục tiêu của Optuna được thiết lập nhằm tìm kiếm bộ siêu tham số tối ưu hóa $F_1\text{-Score}$ kết hợp dò tìm ngưỡng phân loại tốt nhất (thay vì cố định ngưỡng mặc định $0{,}5$). Cụ thể, tại mỗi chu kỳ thử nghiệm, thuật toán đánh giá phân phối xác suất dự đoán $\hat{p}$ trên tập validation và xác định ngưỡng $th^*$ mang lại $F_1\text{-Score}$ cao nhất:
\begin{equation*}
    \text{Score}_{\text{optimal}} = \max_{th \in (0, 1)} F_1\big(y, \, \mathbb{I}(\hat{p} \ge th)\big)
\end{equation*}
Chiến lược này giúp mô hình đạt điểm số tối ưu và phản ánh chính xác hiệu năng thực tế trên bài toán phát hiện bất thường phụ tải.

\paragraph{Phân tích đặc trưng các bộ siêu tham số tối ưu:}\hspace{1cm}

Từ các bộ siêu tham số tối ưu thu được tại Bảng~\ref{tab:tuned_hyperparameters}, có thể rút ra một số nhận định kỹ thuật quan trọng về hành vi của từng mô hình:
\begin{itemize}
    \item \textbf{Cân bằng giữa dung lượng mô hình và điều chuẩn trên LightGBM:} Thuật toán Bayes TPE đã mở rộng số lượng lá (\texttt{num\_leaves} đạt $233$ ở Baseline và $252$ ở Zone) cùng độ sâu cây (\texttt{max\_depth} từ $16 - 17$). Nhờ cơ chế Leaf-wise kết hợp cùng việc kiểm soát \texttt{feature\_fraction} ($0{,}57 - 0{,}85$) và \texttt{min\_child\_samples} ($62 - 73$), LightGBM khai thác sâu các tương tác phi tuyến phức tạp trong dữ liệu mà vẫn duy trì tính tổng quát hóa cao, ngăn ngừa quá khớp.
    \item \textbf{Cơ chế lấy mẫu đặc trưng ngẫu nhiên trên XGBoost:} XGBoost đạt trạng thái tối ưu ở độ sâu cây lớn (\texttt{max\_depth}: $15 - 16$), tốc độ học \texttt{learning\_rate} $\approx 0{,}24$ với việc chỉ lấy mẫu khoảng $60\% - 62\%$ số lượng đặc trưng tại mỗi cây (\texttt{colsample\_bytree}). Cơ chế này giúp giảm tương quan giữa các cây quyết định độc lập, tạo ra các biên phân tách cực kỳ vững chắc và đẩy F1-Score lên đỉnh cao nhất.
    \item \textbf{Điều chuẩn chặt chẽ trên CatBoost:} Thuật toán CatBoost phát huy hiệu quả cao nhất với độ sâu cây đối xứng \texttt{depth}: $14$ và hệ số điều chuẩn $L_2$ (\texttt{l2\_leaf\_reg}) đạt mức $1{,}40$ (Baseline) và $2{,}26$ (Zone). Việc siết chặt điều chuẩn trên nhánh Zone giúp mô hình kiểm soát tốt biên độ lá khi tiếp nhận các đặc trưng tương tác phân vùng, giữ cho phân phối xác suất dự đoán luôn mượt và ổn định.
\end{itemize}
Việc hoàn thiện tinh chỉnh các siêu tham số này tạo tiền đề vững chắc để các mô hình đơn đạt tới trạng thái cực đại hóa phân tách nhãn trước khi tiến hành tối ưu hóa ngưỡng quyết định và dung hợp Ensemble.

\subsubsection{Kỹ thuật Ngưỡng Động 16 Khu vực (Site-Specific Dynamic Thresholding)}

Như đã phân tích và áp dụng nhất quán xuyên suốt các giai đoạn thực nghiệm trên, việc dịch chuyển ngưỡng phân loại toàn cục $\tau^*$ đã chứng minh hiệu quả vượt trội trong việc khắc phục tình trạng mất cân bằng lớp trầm trọng, giúp $F_1$-Score của toàn bộ các mô hình bứt phá so với ngưỡng mặc định $0{,}5$ (minh chứng đối chiếu tại Bảng~\ref{tab:model_feasibility_comparison}).

Tuy nhiên, việc sử dụng một ngưỡng cắt tĩnh duy nhất $\tau^*$ cho toàn bộ hệ thống vẫn chứa đựng một giả định mang tính đơn giản hóa: \textit{coi mọi khu vực địa lý đều có cùng quy luật phụ tải và biên độ dao động bất thường như nhau}.

\paragraph{Cơ sở Lý thuyết: Ngưỡng Động Phi tham số (Nonparametric Dynamic Thresholding):}\hspace{1cm}

Để khắc phục hạn chế này, nhóm nghiên cứu đã tham khảo cơ sở lý thuyết từ công trình nổi tiếng của Hundman et al. (KDD 2018) \textit{"Detecting Spacecraft Anomalies Using LSTMs and Nonparametric Dynamic Thresholding"}. Hundman và các cộng sự chỉ ra rằng trong các hệ thống chuỗi thời gian phân tán phức tạp, mỗi kênh cảm biến hoặc khu vực hoạt động chịu tác động của các điều kiện môi trường riêng biệt; do đó, việc áp đặt một ngưỡng cắt tĩnh duy nhất cho toàn bộ hệ thống là không phù hợp. Ngưỡng phát hiện bất thường cần phải được điều chỉnh động theo phân phối xác suất dự đoán của từng cụm cục bộ.

\paragraph{Triển khai Ngưỡng Động cho 16 Khu vực:}\hspace{1cm}

Áp dụng nguyên lý trên vào bài toán, nhóm thiết lập thuật toán \textbf{Ngưỡng Động theo Khu vực (Dynamic Threshold per Zone)}:
\begin{enumerate}
    \item Dữ liệu kiểm định được phân tách theo 16 khu vực địa lý (\texttt{site\_id} $k \in \{0, \dots, 15\}$).
    \item Với mỗi khu vực $k$, thuật toán xác định một ngưỡng cắt riêng biệt $\tau_k^*$:
    \begin{equation*}
        \tau_k^* = \arg\max_{\tau \in [0{,}01; 0{,}99]} F_1\left(y_{\text{val}}^{(k)}, \mathbb{1}[P(y=1|X_{\text{val}}^{(k)}) \geq \tau]\right)
    \end{equation*}
    Nếu một khu vực không có mẫu dị thường nào trong tập kiểm định, ngưỡng được gán an toàn $\tau_k^* = 1{,}0$.
    \item Khi đưa vào dự đoán thực tế trên tập kiểm tra, mẫu thuộc khu vực nào sẽ được áp dụng ngưỡng cắt tương ứng của khu vực đó:
    \begin{equation*}
        \hat{y}_i = \mathbb{1}\left[P(y_i=1|\mathbf{x}_i) \geq \tau_{\text{site}(i)}^*\right]
    \end{equation*}
\end{enumerate}

Thực nghiệm cho thấy cơ chế ngưỡng động theo 16 khu vực giúp F1-Score tiếp tục cải thiện thêm từ $+0{,}18\%$ đến $+0{,}47\%$ trên các mô hình (tiêu biểu như XGBoost nhánh Zone tăng từ $0{,}9317$ lên $0{,}9341$, LightGBM tăng từ $0{,}9240$ lên $0{,}9287$), khẳng định tính thích ứng cục bộ vượt trội khi đối phó với dữ liệu chuỗi thời gian phụ tải đa phân vùng.

\subsection{Kỹ thuật Dung hợp Mô hình Nâng cao (Model Ensemble)}

\subsubsection{Cơ sở Lý thuyết: Lựa chọn Tập hợp Mô hình (Ensemble Selection)}
Để đẩy hiệu năng phân lớp lên mức tối đa, nhóm nghiên cứu đã áp dụng kỹ thuật \textbf{Weighted Blending} dựa trên cơ sở lý thuyết bài báo kinh điển của Caruana et al. (ICML 2004) \textit{"Ensemble Selection from Libraries of Models"}. Caruana chứng minh rằng việc kết hợp tuyến tính có trọng số từ một tập hợp các mô hình có cấu trúc đa dạng (\textit{diverse architectures}) luôn vượt trội so với bất kỳ mô hình đơn lẻ nào, đồng thời ưu việt hơn phép lấy trung bình cộng đơn thuần nhờ khả năng tối ưu hóa trọng số đóng góp của từng thành phần. Nguyên lý này cũng là chiến thuật nòng cốt được áp dụng rộng rãi và chứng minh tính hiệu quả vượt trội trong các bài toán phân tích chuỗi thời gian phụ tải năng lượng quy mô lớn như \textit{ASHRAE Great Energy Predictor III}.

\subsubsection{Các Phương pháp Dung hợp Thực nghiệm}
Nhằm khảo sát toàn diện các chiến lược kết hợp thông tin, nhóm nghiên cứu đã hiện thực hóa và kiểm định sáu kỹ thuật dung hợp mô hình từ cấp độ tuyến tính đến phi tuyến thích nghi:
\begin{enumerate}
    \item \textbf{Soft Voting (Trung bình cộng xác suất không trọng số):} Phương pháp cơ bản nhất, lấy giá trị trung bình số học giữa các xác suất dự báo:
    \begin{equation*}
        P_{\text{vote}} = \frac{1}{3} \left(P_{\text{XGB}} + P_{\text{LGB}} + P_{\text{Cat}}\right)
    \end{equation*}
    Phương pháp này giúp giảm phương sai dự báo (\textit{variance reduction}) mà không cần tối ưu trọng số, đóng vai trò là mốc chuẩn so sánh cho các cơ chế dung hợp nâng cao.

    \item \textbf{Top-2 Weighted Blending (Tổ hợp 2 mô hình chủ lực XGBoost + CatBoost):} Tổ hợp tuyến tính tập trung vào hai thuật toán có năng lực phân tách tốt nhất và ít tương quan sai số nhất, loại trừ mô hình thứ ba để kiểm định giả thuyết tối giản hóa:
    \begin{equation*}
        P_{\text{top2}} = w_{\text{XGB}} P_{\text{XGB}} + w_{\text{Cat}} P_{\text{Cat}}, \quad \text{với } w_{\text{XGB}} + w_{\text{Cat}} = 1, \; w \ge 0
    \end{equation*}
    Vector trọng số được tối ưu hóa theo hàm mất mát Log-Loss, cho kết quả:
    \begin{itemize}
        \item Trên Baseline: $\mathbf{w} = [0{,}84\,\text{XGB}, \; 0{,}16\,\text{Cat}]$.
        \item Trên Zone: $\mathbf{w} = [0{,}80\,\text{XGB}, \; 0{,}20\,\text{Cat}]$.
    \end{itemize}

    \item \textbf{Tri-Model Weighted Blending LogLoss (Trộn trọng số 3 mô hình tối thiểu hóa Log-Loss):} Tổ hợp lồi của cả ba mô hình với vector trọng số $\mathbf{w} = [w_{\text{XGB}}, w_{\text{LGB}}, w_{\text{Cat}}]$ thỏa mãn $\sum w_m = 1, w_m \geq 0$:
    \begin{equation*}
        P_{\text{blend}} = w_{\text{XGB}} P_{\text{XGB}} + w_{\text{LGB}} P_{\text{LGB}} + w_{\text{Cat}} P_{\text{Cat}}
    \end{equation*}
    Vector $\mathbf{w}$ được tìm kiếm thông qua giải thuật quy hoạch phi tuyến tuần tự (SLSQP) nhằm cực tiểu hóa hàm mất mát Cross-Entropy (Log-Loss) toàn cục trên tập kiểm định:
    \begin{equation*}
        \mathbf{w}^* = \arg\min_{\mathbf{w}} \left( -\frac{1}{N} \sum_{i=1}^N \left[ y_i \ln(P_{\text{blend}, i}) + (1 - y_i) \ln(1 - P_{\text{blend}, i}) \right] \right)
    \end{equation*}
    Kết quả tối ưu thu được phản ánh cấu trúc tương hỗ lý tưởng giữa ba kiến trúc:
    \begin{itemize}
        \item Trên tập dữ liệu Baseline: $\mathbf{w} = [0{,}68\,\text{XGB}, \; 0{,}26\,\text{LGB}, \; 0{,}06\,\text{Cat}]$.
        \item Trên tập dữ liệu Zone: $\mathbf{w} = [0{,}65\,\text{XGB}, \; 0{,}25\,\text{LGB}, \; 0{,}10\,\text{Cat}]$.
    \end{itemize}
    Trong đó, XGBoost nắm giữ vai trò chủ lực ($65\% - 68\%$), LightGBM đóng góp $25\% - 26\%$ giúp làm mượt phân phối xác suất và gia tăng Recall, và CatBoost giữ $6\% - 10\%$ đóng vai trò điều chuẩn chống quá khớp.

    \item \textbf{Dynamic Threshold-Aware Blending (Dung hợp hướng Ngưỡng động 16 Khu vực):} Thay vì tối ưu hóa hàm Log-Loss thuần túy trên toàn bộ dữ liệu, thuật toán quét lưới trực tiếp không gian trọng số $[w_1, w_2, w_3]$ với bước nhảy $0{,}05$ nhằm cực đại hóa chỉ số $F_1$-Score khi áp dụng đồng thời cơ chế dò tìm ngưỡng động theo từng khu vực địa bàn:
    \begin{equation*}
        \mathbf{w}_{\text{dyn}}^* = \arg\max_{\mathbf{w}} F_1\left(y_{\text{val}}, \mathbb{1}[P_{\text{blend}}(\mathbf{w}) \ge \boldsymbol{\tau}_{\text{site}}^*]\right)
    \end{equation*}
    Kết quả tối ưu:
    \begin{itemize}
        \item Trên Baseline: $\mathbf{w} = [0{,}50\,\text{XGB}, \; 0{,}25\,\text{LGB}, \; 0{,}25\,\text{Cat}]$.
        \item Trên Zone: $\mathbf{w} = [0{,}60\,\text{XGB}, \; 0{,}25\,\text{LGB}, \; 0{,}15\,\text{Cat}]$.
    \end{itemize}

    \item \textbf{Stacking với Meta-Learner Tuyến tính (Logistic Regression):} Hiện thực hóa mô hình Stacking hai tầng kinh điển (Wolpert, 1992): Tầng 1 sinh ma trận xác suất ngoại mẫu $[P_{\text{XGB}}, P_{\text{LGB}}, P_{\text{Cat}}]$; Tầng 2 huấn luyện một mô hình Hồi quy Logistic có điều chuẩn $L_2$ trên tập Meta-Train (tách $50\%$ từ tập validation nhằm triệt tiêu rò rỉ dữ liệu). Mô hình học một siêu phẳng phân tách tối ưu:
    \begin{equation*}
        P_{\text{stack}} = \sigma\left(\beta_0 + \beta_1 P_{\text{XGB}} + \beta_2 P_{\text{LGB}} + \beta_3 P_{\text{Cat}}\right)
    \end{equation*}

    \item \textbf{Context-Aware Mixture of Experts (MoE Meta-Learner tích hợp Ngữ cảnh Khu vực):} Kỹ thuật dung hợp phi tuyến bậc cao: Thay vì chỉ dựa vào xác suất đầu ra như Stacking truyền thống, mô hình Meta-Learner (sử dụng LightGBM với siêu tham số tối ưu) tiếp nhận đồng thời cả vector xác suất $[P_{\text{XGB}}, P_{\text{LGB}}, P_{\text{Cat}}]$ và toàn bộ ma trận đặc trưng bối cảnh công trình (\texttt{site\_id}, \texttt{primary\_use}, \texttt{square\_feet}, các biến tương tác phân vùng). Cơ chế này cho phép Meta-Learner tự động định tuyến trọng số thích nghi theo từng đặc thù công trình và điều kiện vi khí hậu cục bộ.
\end{enumerate}

    \item \textbf{Đặc điểm hành vi của Stacking Logistic Regression:} Mô hình Stacking tuyến tính đạt F1 tĩnh rất tốt ($0{,}9339$ trên Zone và $0{,}9290$ trên Baseline). Tuy nhiên, do hàm Sigmoid gặp phải dữ liệu mất cân bằng nghiêm trọng, phân phối xác suất đầu ra của Meta-Learner bị nén mạnh về gần $0$, khiến ngưỡng cắt tối ưu bị kéo sâu xuống $\tau^* = 0{,}068 - 0{,}117$.
    \item \textbf{Năng lực thích ứng của Context-Aware MoE:} Bằng cách đưa thông tin địa bàn \texttt{site\_id} và quy mô công trình vào Meta-Learner LightGBM, mô hình đạt F1 tĩnh $0{,}9320$ và ngưỡng động $0{,}9354$ mà không cần dò tìm ngưỡng quá sâu ($\tau^* = 0{,}294$), thể hiện khả năng định tuyến xác suất thích nghi theo đặc thù từng khu vực.
    \item \textbf{Đỉnh cao Hiệu năng Toàn diện:} Sự kết hợp giữa bộ đặc trưng \textbf{Zone Features}, thuật toán \textbf{Weighted Blending LogLoss [0,65 XGB + 0,25 LGB + 0,10 Cat]} và cơ chế \textbf{Ngưỡng Động 16 Khu vực} đã thiết lập kỷ lục hiệu năng cao nhất của đề tài với $F_1\text{-Score} = \mathbf{0{,}9373}$ (Precision đạt $\mathbf{0{,}9586}$, Recall đạt $\mathbf{0{,}9169}$).
\end{itemize}
