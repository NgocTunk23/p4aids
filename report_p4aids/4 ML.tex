\section{Triển khai thuật toán Học Máy - Machine Learning}

\subsection{Thước đo Đánh giá và Cơ chế Ngưỡng Quyết định}

\subsubsection{Thước đo Đánh giá}
Trong bài toán phát hiện dị thường với tỷ lệ mất cân bằng nghiêm trọng ($97{,}87\%$ bình thường vs $2{,}13\%$ dị thường), việc lựa chọn thước đo đánh giá là yếu tố sống còn để đảm bảo tính khách quan:
\begin{itemize}
    \item \textbf{Vô hiệu hóa Độ chính xác Tổng thể (Accuracy):} Một mô hình ngây thơ dự đoán $100\%$ mẫu là bình thường ($y=0$) vẫn đạt $\text{Accuracy} = 97{,}87\%$, nhưng hoàn toàn thất bại trong việc cảnh báo sự cố rò rỉ hay quá tải điện.
    \item \textbf{Sự sai lệch của ROC-AUC:} Do số lượng mẫu âm tính quá lớn ($TN > 1{,}71$ triệu dòng), tỷ lệ báo động giả $\text{FPR} = \frac{FP}{FP+TN}$ luôn bị ghìm ở mức cực nhỏ dù mô hình dự đoán nhầm hàng chục ngàn mẫu. Điều này tạo ra ảo tưởng lạc quan với điểm $\text{ROC-AUC} > 0{,}95$ nhưng thực chất độ chính xác dự báo dị thường lại rất thấp.
    \item \textbf{F1-Score là Thước đo Đánh giá Tối thượng (Primary Metric):} Mục tiêu vận hành thực tế của hệ thống điện đòi hỏi: vừa phải bắt trúng tối đa các sự cố bất thường để kịp thời xử lý (tối đa hóa Recall), vừa phải giảm thiểu các báo động giả gây lãng phí nhân lực kiểm tra hiện trường (tối đa hóa Precision). $F_1$-Score chính là trung bình điều hòa (\textit{harmonic mean}) ràng buộc chặt chẽ cả hai đại lượng:
    \begin{equation*}
        \text{Precision} = \frac{TP}{TP + FP}, \quad \text{Recall} = \frac{TP}{TP + FN}
    \end{equation*}
    \begin{equation*}
        F_1\text{-Score} = 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}
    \end{equation*}
    Nếu một trong hai chỉ số Precision hoặc Recall bị suy giảm, $F_1$-Score sẽ lập tức bị kéo tụt. Toàn bộ các vòng lặp tối ưu hóa và so sánh mô hình trong chương này đều sử dụng $F_1$-Score làm hàm mục tiêu quyết định.
\end{itemize}

\subsubsection{Cơ chế Ngưỡng Quyết định}\hspace{1cm}
Do nhãn $1$ quá khan hiếm, hàm mất mát Log-loss đẩy xác suất dự đoán $P(y=1)$ lệch mạnh về gần $0$. Nhóm thiết lập cơ chế ra quyết định hai tầng:
\begin{enumerate}
    \item \textbf{Tầng 1 (Dự báo Xác suất Liên tục):} Mô hình tính toán xác suất dị thường liên tục $\hat{p}_i = P(y_i=1 \mid \mathbf{x}_i) \in [0, 1]$.
    \item \textbf{Tầng 2 (Quyết định Gán nhãn qua Ngưỡng Tối ưu):} Nhãn nhị phân được gán thông qua ngưỡng cắt $\tau$:
    \begin{equation*}
        \hat{y}_i = \mathbb{1}\left[\hat{p}_i \ge \tau\right]
    \end{equation*}
    trong đó $\tau$ không cố định ở mức $0{,}5$ mà được quét lưới để cực đại hóa $F_1$-Score trên tập kiểm định:
    \begin{equation*}
        \tau^* = \arg\max_{\tau \in [0{,}01; 0{,}99]} F_1\left(y_{\text{val}}, \, \mathbb{1}[\hat{p}_{\text{val}} \geq \tau]\right)
    \end{equation*}
\end{enumerate}

\subsection{Thử nghiệm, Huấn luyện và Đánh giá các Mô hình}

\subsubsection{Chiến lược Phân chia Dữ liệu và Kiểm soát Phân phối Tập Huấn luyện - Kiểm tra (Train vs. Test Distribution)}

Trước khi tiến hành huấn luyện và so sánh các thuật toán học máy, một bước thiết kế thực nghiệm mang tính chất sống còn là thiết lập phân vùng dữ liệu và thẩm định tính đồng nhất phân phối. Trong nghiên cứu này, toàn bộ quá trình phát triển mô hình sử dụng tập dữ liệu có nhãn đã hoàn tất quy trình tiền xử lý chuẩn hóa (\texttt{train\_preprocessed.parquet} gồm $1.749.494$ bản ghi). Nhằm đánh giá khách quan năng lực tổng quát hóa và ngăn ngừa hiện tượng rò rỉ thông tin (\textit{data leakage}), tập dữ liệu được phân chia thành hai tập con độc lập:
\begin{itemize}
    \item \textbf{Tập Huấn luyện (Train set):} Chiếm tỷ lệ $80\%$ ($1.399.595$ bản ghi), dùng để tối ưu hóa trọng số mô hình và tinh chỉnh siêu tham số.
    \item \textbf{Tập Kiểm tra (Test set):} Chiếm tỷ lệ $20\%$ ($349.899$ bản ghi), được giữ hoàn toàn độc lập và chỉ dùng để đo đạc hiệu năng dự báo thực tế (trong một số ngữ cảnh thử nghiệm nội bộ tập này còn được xem là tập kiểm định ngoại suy \textit{holdout validation}, tuy nhiên nhóm quy ước định danh thống nhất là \textbf{Tập Kiểm tra} theo đúng quy trình chuẩn của bài toán).
\end{itemize}

Để đảm bảo việc phân chia không làm biến dạng bài toán mất cân bằng nhãn nghiêm trọng ($97{,}87\%$ bình thường vs $2{,}13\%$ dị thường), kỹ thuật phân chia ngẫu nhiên phân tầng (\textit{Stratified Partitioning}) với hạt giống cố định (\texttt{random\_state=42}) được bắt buộc áp dụng. Toàn bộ kết quả kiểm toán phân phối giữa tập Huấn luyện và tập Kiểm tra được trực quan hóa tại Hình~\ref{fig:train_test_distribution} và tổng hợp định lượng tại Bảng~\ref{tab:train_test_distribution_audit}.

Phân tích Hình~\ref{fig:train_test_distribution} và Bảng~\ref{tab:train_test_distribution_audit} xác thực 4 kết luận kỹ thuật cốt lõi:
\begin{enumerate}
    \item \textbf{Bảo toàn tuyệt đối tỷ lệ nhãn mục tiêu (Hình~\ref{fig:train_test_distribution}a):} Tỷ lệ lớp dị thường trên tập Huấn luyện đạt chính xác $2{,}1318\%$ ($29.837$ mẫu) và trên tập Kiểm tra cũng đạt đúng $2{,}1318\%$ ($7.459$ mẫu). Mức sai lệch giữa hai tập bằng đúng $0{,}0000\%$, chứng minh chiến lược Stratified Partitioning đã loại bỏ hoàn toàn rủi ro thiên lệch nhãn (\textit{Label Shift}).
    \item \textbf{Loại trừ lệch phân phối đặc trưng (\textit{Covariate Shift}, Hình~\ref{fig:train_test_distribution}):} Các đường ước lượng mật độ xác suất (KDE) của biến phụ tải tiêu thụ \texttt{meter\_reading}, nhiệt độ không khí \texttt{air\_temperature} và diện tích sàn \texttt{square\_feet} giữa hai tập chồng khít lên nhau hoàn hảo. Giá trị thống kê khoảng cách Kolmogorov-Smirnov ($D$) trên toàn bộ các biến đều ở mức rất nhỏ ($D < 0{,}003$). Đặc biệt, kiểm định KS trên các biến khí tượng và công trình đều cho $p\text{-value} > 0{,}05$, xác nhận giả thuyết không ($H_0$) rằng hai tập mẫu xuất phát từ cùng một phân phối xác suất tổng thể.
    \item \textbf{Ngăn ngừa rò rỉ thông tin chuẩn hóa (\textit{Data Leakage Prevention}):} Do toàn bộ các tham số biến đổi (Z-score mean $\mu$, std $\sigma$ và tần suất mã hóa) đều được đóng băng và fit độc lập trên không gian huấn luyện, tập Kiểm tra hoạt động như một thực thể hoàn toàn "vô hình" đối với mô hình trong quá trình học.
    \item \textbf{Tạo lập chuẩn mực đối chứng tin cậy:} Sự đồng nhất phân phối này là bảo chứng khoa học khẳng định: mọi sự vượt trội về chỉ số $F_1$-Score, Precision hay Recall giữa các thuật toán ở các phần tiếp theo hoàn toàn phản ánh năng lực phân tách thực tế của thuật toán, chứ không bị chi phối bởi bất kỳ yếu tố may rủi nào do chia tách dữ liệu.
\end{enumerate}

    \end{itemize}

\textbf{Quyết định kỹ thuật:} Nhóm quyết định loại bỏ hoàn toàn Random Forest và Extra Trees khỏi các giai đoạn tinh chỉnh tiếp theo và khâu dung hợp Ensemble, tập trung $100\%$ tài nguyên tính toán vào ba thuật toán Gradient Boosting vượt trội: \textbf{XGBoost}, \textbf{LightGBM} và \textbf{CatBoost}.

\paragraph{Thử nghiệm và bác bỏ điểm dị biệt từ Isolation Forest (Unsupervised Meta-Feature):}\hspace{1cm}

\textbf{Phân tích và Nhận xét Thực nghiệm chuyên sâu:} 
Bảng~\ref{tab:lag_vs_zone_leakage_test} đem lại góc nhìn đối chiếu đa chiều giữa ba chiến lược đặc trưng trên cùng bộ siêu tham số đã tối ưu:
\begin{itemize}
    \item \textbf{Tác động tiêu cực của Đặc trưng Trễ (Lag Features) so với Baseline:} Khi đưa các biến trễ phụ tải thô (\texttt{meter\_}\allowbreak\texttt{reading} tại các bước $t-1, t-2$) vào mô hình, hiệu năng của cả ba thuật toán đều bị suy giảm so với mốc Baseline chuẩn. Đáng chú ý nhất, F1-Score của LightGBM sụt giảm mạnh từ $0{,}9284$ xuống $0{,}8964$ (giảm $-3{,}20\%$) và CatBoost giảm từ $0{,}9200$ xuống $0{,}9107$ (giảm $-0{,}93\%$). Nguyên nhân là do trong chuỗi dữ liệu thực tế, các khoảng trống mất tín hiệu cảm biến hoặc biến động bất thường đột ngột làm đứt đoạn chuỗi trễ, biến các giá trị trễ thành nguồn phát tán nhiễu thay vì cung cấp tín hiệu dự báo tin cậy.
    \item \textbf{Sự ưu việt vượt trội của Đặc trưng Khu vực (Zone Features) trong Dung hợp Ensemble:} Việc bổ sung các biến tương tác bối cảnh khu vực (\texttt{Zone Features} gồm tương tác giữa \texttt{site\_id}, \allowbreak\texttt{primary\_use} và \allowbreak\texttt{square\_feet}) kết hợp mã hóa One-Hot trực giao mang lại sự ổn định và độ nhạy phát hiện bất thường cao độ:
    \begin{itemize}
        \item Trên từng mô hình đơn lẻ, Zone Features duy trì độ chính xác Precision cực cao ($0{,}9520 - 0{,}9612$) và độ nhạy Recall vững chắc ($88{,}95\% - 90{,}79\%$).
        \item Đặc biệt, khi bước vào khâu dung hợp đa mô hình kết hợp Ngưỡng động 16 khu vực, Zone Features đã giúp tổ hợp Weighted Blending bứt phá lên đỉnh kỷ lục tuyệt đối toàn đề tài với $F_1 = \mathbf{0{,}9373}$ (vượt trội so với mức $0{,}9365$ của Baseline).
    \end{itemize}
    \item \textbf{Tính bền vững và tính sẵn sàng trong triển khai thực tế:} Nhóm đặc trưng Zone Features khai thác các thuộc tính tĩnh về địa bàn công trình và quy mô diện tích. Đây là các thông tin bối cảnh luôn sẵn sàng $100\%$, không đòi hỏi đồng bộ chuỗi thời gian liên tục hay lo ngại mất gói tin mạng IoT. Điều này vừa giúp nâng cao độ chính xác phân loại, vừa bảo đảm an toàn vận hành, loại bỏ hoàn toàn nguy cơ sụp đổ dây chuyền do mất dữ liệu lịch sử.
\end{itemize}
Do đó, nhánh \textbf{Zone Features} được nhóm nghiên cứu tin tưởng lựa chọn làm bộ đặc trưng nòng cốt duy nhất để tiến hành dung hợp các mô hình Ensemble tiếp theo.

\paragraph{Chi tiết Định nghĩa, Cơ chế Mã hóa và Dữ liệu Mẫu của Nhóm Đặc trưng Zone:}\hspace{1cm}

Nhóm đặc trưng tương tác khu vực (\texttt{Zone Features}) được cấu tạo từ 3 biến phái sinh nhằm mô hình hóa sự khác biệt về địa bàn và quy mô phụ tải:
\begin{enumerate}
    \item \texttt{zone\_interaction}: Mã hóa tổ hợp định danh phân vùng giữa mã khu vực \texttt{site\_id} và mục đích sử dụng công trình \texttt{primary\_use}. Biến số này được tạo qua 3 bước:
    \begin{itemize}
        \item \textit{Bước 1:} Mã hóa danh mục \texttt{primary\_use} thành chỉ số nguyên $\text{code} \in \{0, 1, \dots, 15\}$.
        \item \textit{Bước 2:} Ghép chuỗi tạo khóa tổ hợp đại diện: $\text{Key} = \texttt{site\_id} \,\|\, \text{``\_''} \,\|\, \text{code}$ (ví dụ: \texttt{"0\_0"} cho Trường học tại Site 0, \texttt{"0\_6"} cho Văn phòng tại Site 0).
        \item \textit{Bước 3:} Áp dụng Label Encoding gán số nguyên định danh cho $47$ cặp tổ hợp phân vùng thực tế:
        \begin{equation}
            \texttt{zone\_interaction} = \text{CategoricalCode}\Big(\texttt{site\_id} \,\|\, \text{``\_''} \,\|\, \text{code}\Big) \in \{0, 1, \dots, 46\}
        \end{equation}
    \end{itemize}
    \item \texttt{site\_sqft\_mean}: Diện tích sàn trung bình của toàn bộ các công trình thuộc cùng khu vực địa bàn $S_i$ (chuẩn hóa Z-score):
    \begin{equation}
        \texttt{site\_sqft\_mean} = \frac{1}{|S_i|} \sum_{b \in S_i} \text{Z}(\texttt{square\_feet}_b)
    \end{equation}
    \item \texttt{use\_sqft\_mean}: Diện tích sàn trung bình của toàn bộ các công trình có cùng mục đích sử dụng $U_j$ trên toàn hệ thống (chuẩn hóa Z-score):
    \begin{equation}
        \texttt{use\_sqft\_mean} = \frac{1}{|U_j|} \sum_{b \in U_j} \text{Z}(\texttt{square\_feet}_b)
    \end{equation}
\end{enumerate}

\paragraph{Vấn đề Quan hệ Thứ tự Giả định ($3 < 4$) khi Mã hóa \texttt{zone\_interaction}:}\hspace{1cm}

Một vấn đề kỹ thuật quan trọng thường được đặt ra trong học máy là: \textit{Liệu việc gán nhãn số nguyên $0, 1, 2, \dots, 46$ có khiến mô hình ngộ nhận một quan hệ thứ bậc giả định (ví dụ: lớp 3 bé hơn lớp 4 hay lớp 4 gấp đôi lớp 2) hay không?} Nhóm nghiên cứu khẳng định điều này hoàn toàn không xảy ra trong đề tài nhờ các cơ sở khoa học sau:
\begin{itemize}
    \item \textbf{Bản chất phân nhánh không phụ thuộc độ lớn của Cây Quyết định:} Trong các mô hình tuyến tính ($y = \sum w_i x_i$), số nguyên liên tiếp sẽ bị nhân trực tiếp với trọng số $w_i$, gây ra sai lệch thứ bậc. Ngược lại, toàn bộ các mô hình sử dụng trong nhánh Zone đều là họ cây Boosting (XGBoost, LightGBM, CatBoost). Cây quyết định chỉ thực hiện các phép so sánh ngưỡng nhị phân $x \le c$ hoặc $x > c$. Với cấu hình độ sâu tối ưu ($\texttt{max\_depth} = 15 - 16$), cây có thể dễ dàng cô lập bất kỳ giá trị phân vùng đơn lẻ nào qua các phép cắt nhị phân liên tiếp mà không bị chi phối bởi khoảng cách hay độ lớn đại số giữa các số nguyên.
    \item \textbf{Cơ chế Gom cụm Danh mục Tối ưu của LightGBM và CatBoost:} Cả LightGBM và CatBoost đều tích hợp thuật toán phân chia biến phân loại chuyên biệt (\textit{optimal categorical partitioning}): thuật toán sắp xếp các nhóm danh mục dựa trên gradient và phân chia thành các tập con $x \in S$ so với $x \notin S$ (thay vì so sánh $x < c$ thông thường). Điều này loại bỏ hoàn toàn tính thứ tự của mã hóa số nguyên.
    \item \textbf{Duy trì tính trực giao với Không gian One-Hot Gốc:} Song song với \texttt{zone\_interaction}, toàn bộ 16 thuộc tính công năng gốc vẫn được duy trì đồng thời dưới dạng 16 cột nhị phân One-Hot Encoding (\texttt{primary\_use\_Education}, \texttt{primary\_use\_Office}, \dots). Do đó, \texttt{zone\_interaction} thuần túy đóng vai trò là một đặc trưng phụ trợ giúp các cây quyết định tăng tốc độ tìm kiếm lát cắt cục bộ mà không làm mất đi tính độc lập không gian đặc trưng.
\end{itemize}

\paragraph{Luận cứ Khoa học: Vì sao Áp dụng Mã hóa Số nguyên cho Baseline và One-Hot cho Zone?}\hspace{1cm}

Một vấn đề mang tính quyết định về mặt kỹ thuật đặc trưng là: \textit{Tại sao cùng là thuộc tính mục đích sử dụng công trình (\texttt{primary\_use}), nhóm nghiên cứu lại áp dụng mã hóa số nguyên đơn biến (\texttt{primary\_use\_code}) cho nhánh Baseline nhưng lại duy trì mã hóa One-Hot ($13$ cột nhị phân) cho nhánh Zone?}

Quyết định kiến trúc này được bảo chứng bởi các phân tích thực nghiệm và lý thuyết học máy vững chắc:
\begin{enumerate}
    \item \textbf{Đối với Nhánh Baseline (Loại trừ ma trận thưa và hiện tượng ``mù đặc trưng''):}
    \begin{itemize}
        \item Ở nhánh Baseline, hệ thống không có sự trợ lực của đặc trưng tương tác phân vùng \texttt{zone\_interaction}.
        \item Nếu sử dụng One-Hot, $13$ cột nhị phân chiếm tới $25\%$ không gian đặc trưng ($13/52$ cột) nhưng đa số giá trị là $0$ (ma trận thưa). Do các thuật toán Gradient Boosting sử dụng kỹ thuật lấy mẫu ngẫu nhiên tập con đặc trưng tại mỗi cây (\texttt{colsample\_bytree} $= 0{,}62$ ở XGBoost và \texttt{feature\_fraction} $= 0{,}84$ ở LightGBM), một cây con rất dễ chỉ bốc phải $1 - 2$ cột One-Hot mang giá trị $0$, dẫn tới việc cây hoàn toàn bị ``mù'' thông tin về công năng công trình.
        \item Khi chuyển sang \textbf{1 cột số nguyên \texttt{primary\_use\_code}}, cây chỉ cần chọn trúng cột này là nắm trọn $100\%$ thông tin công năng trong duy nhất $1$ phép phân nhánh. Thực nghiệm đối chứng cho thấy: trên tập Baseline mới ($40$ đặc trưng), $F_1$-Score của XGBoost tăng từ $0{,}9282$ lên $0{,}9294$ ($+0{,}13\%$), LightGBM tăng từ $0{,}9191$ lên $0{,}9212$ ($+0{,}23\%$), Precision tăng vọt lên mức kỷ lục $96{,}09\% - 96{,}42\%$, đồng thời thời gian huấn luyện rút ngắn $8 - 10$ giây.
    \end{itemize}
    \item \textbf{Đối với Nhánh Zone (Tương hỗ Đa chiều giữa One-Hot và Tương tác Phân vùng):}
    \begin{itemize}
        \item Ở nhánh Zone, biến tổ hợp \texttt{zone\_interaction} ($\text{site\_id} \times \text{primary\_use}$) đã đảm nhiệm việc định vị chính xác hành vi phụ tải cục bộ của từng loại công trình tại từng vùng khí hậu.
        \item Khi đó, việc duy trì $13$ cột One-Hot trực giao không còn gây nhiễu, mà ngược lại, cung cấp góc nhìn đối chiếu toàn cục: cho phép cây quyết định so sánh hành vi của cùng một loại công trình (ví dụ: toàn bộ trường học \texttt{primary\_use\_Education} $= 1$) xuyên suốt các vùng địa lý khác nhau. Sự kết hợp cộng hưởng này giúp mô hình tổ hợp Weighted Blending trên tập Zone bứt phá lên mốc kỷ lục toàn đề tài $F_1 = \mathbf{0{,}9373}$.
    \end{itemize}
\end{enumerate}

\subsubsection{Tinh chỉnh Siêu tham số (Hyperparameter Tuning) với Optuna}

Nhằm khai thác tối đa tiềm năng của từng thuật toán, nhóm sử dụng khung tối ưu hóa \textbf{Optuna} dựa trên thuật toán Bayes TPE (\textit{Tree-structured Parzen Estimator}). Không gian tìm kiếm được cấu hình riêng cho từng thuật toán qua các chu kỳ thử nghiệm (\textit{trials}).

Do dữ liệu có mức độ mất cân bằng lớp cao và thang đo đánh giá cốt lõi của bài toán là chỉ số $F_1$, hàm mục tiêu của Optuna được thiết lập nhằm tìm kiếm bộ siêu tham số tối ưu hóa $F_1\text{-Score}$ kết hợp dò tìm ngưỡng phân loại tốt nhất (thay vì cố định ngưỡng mặc định $0{,}5$). Cụ thể, tại mỗi chu kỳ thử nghiệm, thuật toán đánh giá phân phối xác suất dự đoán $\hat{p}$ trên tập validation và xác định ngưỡng $th^*$ mang lại $F_1\text{-Score}$ cao nhất:
\begin{equation*}
    \text{Score}_{\text{optimal}} = \max_{th \in (0, 1)} F_1\big(y, \, \mathbb{I}(\hat{p} \ge th)\big)
\end{equation*}
Chiến lược này giúp mô hình đạt điểm số tối ưu và phản ánh chính xác hiệu năng thực tế trên bài toán phát hiện bất thường phụ tải.

\paragraph{Phân tích đặc trưng các bộ siêu tham số tối ưu:}\hspace{1cm}

Từ các bộ siêu tham số tối ưu thu được tại Bảng~\ref{tab:tuned_hyperparameters}, có thể rút ra một số nhận định kỹ thuật quan trọng về hành vi của từng mô hình:
\begin{itemize}
    \item \textbf{Cân bằng giữa dung lượng mô hình và điều chuẩn trên LightGBM:} Thuật toán Bayes TPE đã mở rộng số lượng lá (\texttt{num\_leaves} đạt $233$ ở Baseline và $252$ ở Zone) cùng độ sâu cây (\texttt{max\_depth} từ $16 - 17$). Nhờ cơ chế Leaf-wise kết hợp cùng việc kiểm soát \texttt{feature\_fraction} ($0{,}57 - 0{,}85$) và \texttt{min\_child\_samples} ($62 - 73$), LightGBM khai thác sâu các tương tác phi tuyến phức tạp trong dữ liệu mà vẫn duy trì tính tổng quát hóa cao, ngăn ngừa quá khớp.
    \item \textbf{Cơ chế lấy mẫu đặc trưng ngẫu nhiên trên XGBoost:} XGBoost đạt trạng thái tối ưu ở độ sâu cây lớn (\texttt{max\_depth}: $15 - 16$), tốc độ học \texttt{learning\_rate} $\approx 0{,}24$ với việc chỉ lấy mẫu khoảng $60\% - 62\%$ số lượng đặc trưng tại mỗi cây (\texttt{colsample\_bytree}). Cơ chế này giúp giảm tương quan giữa các cây quyết định độc lập, tạo ra các biên phân tách cực kỳ vững chắc và đẩy F1-Score lên đỉnh cao nhất.
    \item \textbf{Điều chuẩn chặt chẽ trên CatBoost:} Thuật toán CatBoost phát huy hiệu quả cao nhất với độ sâu cây đối xứng \texttt{depth}: $14$ và hệ số điều chuẩn $L_2$ (\texttt{l2\_leaf\_reg}) đạt mức $1{,}40$ (Baseline) và $2{,}26$ (Zone). Việc siết chặt điều chuẩn trên nhánh Zone giúp mô hình kiểm soát tốt biên độ lá khi tiếp nhận các đặc trưng tương tác phân vùng, giữ cho phân phối xác suất dự đoán luôn mượt và ổn định.
\end{itemize}
Việc hoàn thiện tinh chỉnh các siêu tham số này tạo tiền đề vững chắc để các mô hình đơn đạt tới trạng thái cực đại hóa phân tách nhãn trước khi tiến hành tối ưu hóa ngưỡng quyết định và dung hợp Ensemble.

\subsubsection{Kỹ thuật Ngưỡng Động 16 Khu vực (Site-Specific Dynamic Thresholding)}

Như đã phân tích và áp dụng nhất quán xuyên suốt các giai đoạn thực nghiệm trên, việc dịch chuyển ngưỡng phân loại toàn cục $\tau^*$ đã chứng minh hiệu quả vượt trội trong việc khắc phục tình trạng mất cân bằng lớp trầm trọng, giúp $F_1$-Score của toàn bộ các mô hình bứt phá so với ngưỡng mặc định $0{,}5$ (minh chứng đối chiếu tại Bảng~\ref{tab:model_feasibility_comparison}).

Tuy nhiên, việc sử dụng một ngưỡng cắt tĩnh duy nhất $\tau^*$ cho toàn bộ hệ thống vẫn chứa đựng một giả định mang tính đơn giản hóa: \textit{coi mọi khu vực địa lý đều có cùng quy luật phụ tải và biên độ dao động bất thường như nhau}.

\paragraph{Cơ sở Lý thuyết: Ngưỡng Động Phi tham số (Nonparametric Dynamic Thresholding):}\hspace{1cm}

Để khắc phục hạn chế này, nhóm nghiên cứu đã tham khảo cơ sở lý thuyết từ công trình nổi tiếng của Hundman et al. (KDD 2018) \textit{"Detecting Spacecraft Anomalies Using LSTMs and Nonparametric Dynamic Thresholding"}. Hundman và các cộng sự chỉ ra rằng trong các hệ thống chuỗi thời gian phân tán phức tạp, mỗi kênh cảm biến hoặc khu vực hoạt động chịu tác động của các điều kiện môi trường riêng biệt; do đó, việc áp đặt một ngưỡng cắt tĩnh duy nhất cho toàn bộ hệ thống là không phù hợp. Ngưỡng phát hiện bất thường cần phải được điều chỉnh động theo phân phối xác suất dự đoán của từng cụm cục bộ.

\paragraph{Triển khai Ngưỡng Động cho 16 Khu vực:}\hspace{1cm}

Áp dụng nguyên lý trên vào bài toán, nhóm thiết lập thuật toán \textbf{Ngưỡng Động theo Khu vực (Dynamic Threshold per Zone)}:
\begin{enumerate}
    \item Dữ liệu kiểm định được phân tách theo 16 khu vực địa lý (\texttt{site\_id} $k \in \{0, \dots, 15\}$).
    \item Với mỗi khu vực $k$, thuật toán xác định một ngưỡng cắt riêng biệt $\tau_k^*$:
    \begin{equation*}
        \tau_k^* = \arg\max_{\tau \in [0{,}01; 0{,}99]} F_1\left(y_{\text{val}}^{(k)}, \mathbb{1}[P(y=1|X_{\text{val}}^{(k)}) \geq \tau]\right)
    \end{equation*}
    Nếu một khu vực không có mẫu dị thường nào trong tập kiểm định, ngưỡng được gán an toàn $\tau_k^* = 1{,}0$.
    \item Khi đưa vào dự đoán thực tế trên tập kiểm tra, mẫu thuộc khu vực nào sẽ được áp dụng ngưỡng cắt tương ứng của khu vực đó:
    \begin{equation*}
        \hat{y}_i = \mathbb{1}\left[P(y_i=1|\mathbf{x}_i) \geq \tau_{\text{site}(i)}^*\right]
    \end{equation*}
\end{enumerate}

Thực nghiệm cho thấy cơ chế ngưỡng động theo 16 khu vực giúp F1-Score tiếp tục cải thiện thêm từ $+0{,}18\%$ đến $+0{,}47\%$ trên các mô hình (tiêu biểu như XGBoost nhánh Zone tăng từ $0{,}9317$ lên $0{,}9341$, LightGBM tăng từ $0{,}9240$ lên $0{,}9287$), khẳng định tính thích ứng cục bộ vượt trội khi đối phó với dữ liệu chuỗi thời gian phụ tải đa phân vùng.

\subsection{Kỹ thuật Dung hợp Mô hình Nâng cao (Model Ensemble)}

\subsubsection{Cơ sở Lý thuyết: Lựa chọn Tập hợp Mô hình (Ensemble Selection)}
Để đẩy hiệu năng phân lớp lên mức tối đa, nhóm nghiên cứu đã áp dụng kỹ thuật \textbf{Weighted Blending} dựa trên cơ sở lý thuyết bài báo kinh điển của Caruana et al. (ICML 2004) \textit{"Ensemble Selection from Libraries of Models"}. Caruana chứng minh rằng việc kết hợp tuyến tính có trọng số từ một tập hợp các mô hình có cấu trúc đa dạng (\textit{diverse architectures}) luôn vượt trội so với bất kỳ mô hình đơn lẻ nào, đồng thời ưu việt hơn phép lấy trung bình cộng đơn thuần nhờ khả năng tối ưu hóa trọng số đóng góp của từng thành phần. Nguyên lý này cũng là chiến thuật nòng cốt được áp dụng rộng rãi và chứng minh tính hiệu quả vượt trội trong các bài toán phân tích chuỗi thời gian phụ tải năng lượng quy mô lớn như \textit{ASHRAE Great Energy Predictor III}.

\subsubsection{Các Phương pháp Dung hợp Thực nghiệm}
Nhằm khảo sát toàn diện các chiến lược kết hợp thông tin, nhóm nghiên cứu đã hiện thực hóa và kiểm định sáu kỹ thuật dung hợp mô hình từ cấp độ tuyến tính đến phi tuyến thích nghi:
\begin{enumerate}
    \item \textbf{Soft Voting (Trung bình cộng xác suất không trọng số):} Phương pháp cơ bản nhất, lấy giá trị trung bình số học giữa các xác suất dự báo:
    \begin{equation*}
        P_{\text{vote}} = \frac{1}{3} \left(P_{\text{XGB}} + P_{\text{LGB}} + P_{\text{Cat}}\right)
    \end{equation*}
    Phương pháp này giúp giảm phương sai dự báo (\textit{variance reduction}) mà không cần tối ưu trọng số, đóng vai trò là mốc chuẩn so sánh cho các cơ chế dung hợp nâng cao.

    \item \textbf{Top-2 Weighted Blending (Tổ hợp 2 mô hình chủ lực XGBoost + CatBoost):} Tổ hợp tuyến tính tập trung vào hai thuật toán có năng lực phân tách tốt nhất và ít tương quan sai số nhất, loại trừ mô hình thứ ba để kiểm định giả thuyết tối giản hóa:
    \begin{equation*}
        P_{\text{top2}} = w_{\text{XGB}} P_{\text{XGB}} + w_{\text{Cat}} P_{\text{Cat}}, \quad \text{với } w_{\text{XGB}} + w_{\text{Cat}} = 1, \; w \ge 0
    \end{equation*}
    Vector trọng số được tối ưu hóa theo hàm mất mát Log-Loss, cho kết quả:
    \begin{itemize}
        \item Trên Baseline: $\mathbf{w} = [0{,}84\,\text{XGB}, \; 0{,}16\,\text{Cat}]$.
        \item Trên Zone: $\mathbf{w} = [0{,}80\,\text{XGB}, \; 0{,}20\,\text{Cat}]$.
    \end{itemize}

    \item \textbf{Tri-Model Weighted Blending LogLoss (Trộn trọng số 3 mô hình tối thiểu hóa Log-Loss):} Tổ hợp lồi của cả ba mô hình với vector trọng số $\mathbf{w} = [w_{\text{XGB}}, w_{\text{LGB}}, w_{\text{Cat}}]$ thỏa mãn $\sum w_m = 1, w_m \geq 0$:
    \begin{equation*}
        P_{\text{blend}} = w_{\text{XGB}} P_{\text{XGB}} + w_{\text{LGB}} P_{\text{LGB}} + w_{\text{Cat}} P_{\text{Cat}}
    \end{equation*}
    Vector $\mathbf{w}$ được tìm kiếm thông qua giải thuật quy hoạch phi tuyến tuần tự (SLSQP) nhằm cực tiểu hóa hàm mất mát Cross-Entropy (Log-Loss) toàn cục trên tập kiểm định:
    \begin{equation*}
        \mathbf{w}^* = \arg\min_{\mathbf{w}} \left( -\frac{1}{N} \sum_{i=1}^N \left[ y_i \ln(P_{\text{blend}, i}) + (1 - y_i) \ln(1 - P_{\text{blend}, i}) \right] \right)
    \end{equation*}
    Kết quả tối ưu thu được phản ánh cấu trúc tương hỗ lý tưởng giữa ba kiến trúc:
    \begin{itemize}
        \item Trên tập dữ liệu Baseline: $\mathbf{w} = [0{,}50\,\text{XGB}, \; 0{,}35\,\text{LGB}, \; 0{,}15\,\text{Cat}]$.
        \item Trên tập dữ liệu Zone: $\mathbf{w} = [0{,}65\,\text{XGB}, \; 0{,}25\,\text{LGB}, \; 0{,}10\,\text{Cat}]$.
    \end{itemize}
    Trong đó, XGBoost nắm giữ vai trò chủ lực ($50\% - 65\%$), LightGBM đóng góp $25\% - 35\%$ giúp làm mượt phân phối xác suất và gia tăng Recall, và CatBoost giữ $10\% - 15\%$ đóng vai trò điều chuẩn chống quá khớp.

    \item \textbf{Dynamic Threshold-Aware Blending (Dung hợp hướng Ngưỡng động 16 Khu vực):} Thay vì tối ưu hóa hàm Log-Loss thuần túy trên toàn bộ dữ liệu, thuật toán quét lưới trực tiếp không gian trọng số $[w_1, w_2, w_3]$ với bước nhảy $0{,}05$ nhằm cực đại hóa chỉ số $F_1$-Score khi áp dụng đồng thời cơ chế dò tìm ngưỡng động theo từng khu vực địa bàn:
    \begin{equation*}
        \mathbf{w}_{\text{dyn}}^* = \arg\max_{\mathbf{w}} F_1\left(y_{\text{val}}, \mathbb{1}[P_{\text{blend}}(\mathbf{w}) \ge \boldsymbol{\tau}_{\text{site}}^*]\right)
    \end{equation*}
    Kết quả tối ưu:
    \begin{itemize}
        \item Trên Baseline: $\mathbf{w} = [0{,}50\,\text{XGB}, \; 0{,}30\,\text{LGB}, \; 0{,}20\,\text{Cat}]$.
        \item Trên Zone: $\mathbf{w} = [0{,}60\,\text{XGB}, \; 0{,}25\,\text{LGB}, \; 0{,}15\,\text{Cat}]$.
    \end{itemize}

    \item \textbf{Stacking với Meta-Learner Tuyến tính (Logistic Regression):} Hiện thực hóa mô hình Stacking hai tầng kinh điển (Wolpert, 1992): Tầng 1 sinh ma trận xác suất ngoại mẫu $[P_{\text{XGB}}, P_{\text{LGB}}, P_{\text{Cat}}]$; Tầng 2 huấn luyện một mô hình Hồi quy Logistic có điều chuẩn $L_2$ trên tập Meta-Train (tách $50\%$ từ tập validation nhằm triệt tiêu rò rỉ dữ liệu). Mô hình học một siêu phẳng phân tách tối ưu:
    \begin{equation*}
        P_{\text{stack}} = \sigma\left(\beta_0 + \beta_1 P_{\text{XGB}} + \beta_2 P_{\text{LGB}} + \beta_3 P_{\text{Cat}}\right)
    \end{equation*}

    \item \textbf{Context-Aware Mixture of Experts (MoE Meta-Learner tích hợp Ngữ cảnh Khu vực):} Kỹ thuật dung hợp phi tuyến bậc cao: Thay vì chỉ dựa vào xác suất đầu ra như Stacking truyền thống, mô hình Meta-Learner (sử dụng LightGBM với siêu tham số tối ưu) tiếp nhận đồng thời cả vector xác suất $[P_{\text{XGB}}, P_{\text{LGB}}, P_{\text{Cat}}]$ và toàn bộ ma trận đặc trưng bối cảnh công trình (\texttt{site\_id}, \texttt{primary\_use}, \texttt{square\_feet}, các biến tương tác phân vùng). Cơ chế này cho phép Meta-Learner tự động định tuyến trọng số thích nghi theo từng đặc thù công trình và điều kiện vi khí hậu cục bộ.
\end{enumerate}

Tuy nhiên, do hàm Sigmoid gặp phải dữ liệu mất cân bằng nghiêm trọng, phân phối xác suất đầu ra của Meta-Learner bị nén mạnh về gần $0$, khiến ngưỡng cắt tối ưu bị kéo sâu xuống $\tau^* = 0{,}068 - 0{,}117$.
    \item \textbf{Năng lực thích ứng của Context-Aware MoE:} Bằng cách đưa thông tin địa bàn \texttt{site\_id} và quy mô công trình vào Meta-Learner LightGBM, mô hình đạt F1 tĩnh $0{,}9320$ và ngưỡng động $0{,}9354$ mà không cần dò tìm ngưỡng quá sâu ($\tau^* = 0{,}294$), thể hiện khả năng định tuyến xác suất thích nghi theo đặc thù từng khu vực.

\end{itemize}
