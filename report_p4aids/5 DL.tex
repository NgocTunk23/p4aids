\section{Triển khai Học Sâu}

\subsection{Thiết kế Pipeline Học Sâu Tổng thể}
Phần Học sâu được hiện thực trong thư mục \texttt{model\_DL/} (các tệp \texttt{config.py}, \texttt{models.py}, \texttt{prepare\_chunked\_data.py}, \texttt{chunked\_data.py}, \texttt{chunked\_pipeline.py}). Pipeline gồm bốn giai đoạn, cùng khung với phần Học máy (tiền xử lý $\rightarrow$ huấn luyện từng mô hình $\rightarrow$ tối ưu $\rightarrow$ dung hợp) để hai phần có thể đặt cạnh nhau. Khác với phiên bản đầu của pipeline (dựng lại lưới giờ từ dữ liệu thô), phiên bản cuối \textbf{dùng chung ba bộ dữ liệu đã chia theo mùa vụ của phần Học máy} (Baseline, Zone, Context; Chương~3) và chỉ gắn thêm cột \texttt{timestamp}, nên mọi mô hình Học sâu và Học máy được chấm trên cùng $372.993$ dòng ngoài mẫu $T_{\text{out}}$. Mỗi giai đoạn ghi rõ nội dung \textbf{đã triển khai} trong mã nguồn và nội dung \textbf{chưa triển khai} (hướng mở rộng), để báo cáo không mô tả quá những gì đã chạy.

Pipeline tuân theo bốn nguyên tắc thiết kế:
\begin{enumerate}
    \item \textbf{Hai lớp đánh giá tách biệt:} (i) trong $T_{\text{in}}$ (ngày $1$--$20$ hằng tháng), mọi mô hình được kiểm định bằng \texttt{GroupKFold} $5$ phần theo \texttt{building\_id}, vì các giờ liền kề của cùng một toà nhà rất giống nhau và chia ngẫu nhiên theo dòng sẽ làm kết quả lạc quan; (ii) $T_{\text{out}}$ (ngày $25$ đến hết tháng, cách $T_{\text{in}}$ khoảng đệm $96$ giờ) chỉ dùng để chấm điểm cuối, không dùng để huấn luyện, chọn checkpoint hay chọn ngưỡng. Lưu ý $T_{\text{out}}$ chứa cùng $200$ toà nhà với $T_{\text{in}}$ nhưng ở các ngày sau, nên đây là phép kiểm tra \textbf{theo thời gian}, không phải toà nhà chưa từng thấy.
    \item \textbf{Chỉ dùng đặc trưng đã tiền xử lý không rò rỉ và chuẩn hoá lại trong từng fold:} đặc trưng chunked đã được khớp thống kê trên $T_{\text{in}}$; thêm vào đó, trung vị, trung bình và độ lệch chuẩn của mỗi kênh được ước lượng lại trên các toà nhà huấn luyện của fold đó.
    \item \textbf{Đầu ra là điểm liên tục:} mọi mô hình trả về điểm bất thường theo từng giờ (xác suất hoặc sai số tái tạo), không nhị phân hoá; ngưỡng chỉ được áp dụng khi tính F1.
    \item \textbf{Khảo sát nhiều hệ hình:} một mô hình theo điểm (MLP) làm đối chứng, ba bộ gán nhãn chuỗi có giám sát (TCN, BiGRU, BiLSTM) và một mô hình tái tạo không giám sát (LSTM-AE), mỗi mô hình chạy trên cả ba biến thể dữ liệu.
\end{enumerate}

\begin{enumerate}
    \item \textbf{Giai đoạn 1: Chuẩn bị dữ liệu chunked, Kiểm định và Dựng chuỗi}
    \begin{itemize}
        \item \textbf{Đã triển khai:} \texttt{prepare\_chunked\_data.py} đọc \texttt{train\_features.csv} chỉ để lấy \texttt{timestamp}, kiểm tra thứ tự dòng khớp với \texttt{building\_id}, \texttt{anomaly} và sáu đặc trưng thời gian, rồi ghi sáu tệp \texttt{*\_chunked\_\{baseline,zone,context\}\_timestamp.parquet} vào \texttt{data/processed\_DL/}. Mọi cột và giá trị gốc được giữ nguyên (chỉ thêm một cột); mã băm SHA-256 và số dòng được ghi trong \texttt{chunked\_timestamp\_metadata.json} và được kiểm tra lại mỗi lần nạp.
        \item \textbf{Đã triển khai:} \texttt{chunked\_data.py} kiểm tra số dòng và số cột khớp bản kê, không có giá trị rỗng ở \texttt{building\_id}, \texttt{timestamp}, \texttt{anomaly}, không trùng cặp (toà nhà, giờ), nhãn nhị phân, mọi giá trị số hữu hạn, và các dòng train chỉ rơi vào ngày $\leq 20$, test chỉ vào ngày $\geq 25$.
        \item \textbf{Đã triển khai:} cắt mỗi toà nhà thành các đoạn \textit{đích} dài $168$ giờ (một tuần), bước nhảy $168$, kèm tối đa $252$ giờ \textit{ngữ cảnh} đứng trước. Ngữ cảnh \textbf{không vượt qua chỗ đứt giờ}: vì $T_{\text{in}}$ và $T_{\text{out}}$ là các khối ngày trong từng tháng, chuỗi của mỗi toà nhà gồm nhiều đoạn liên tục, và cửa sổ được đặt lại ở mỗi chỗ đứt (đệm số $0$ ở đầu cửa sổ). Mỗi giờ là đích đúng một lần nên mỗi dòng được chấm điểm một lần; mã kiểm tra và báo lỗi nếu có dòng bị bỏ sót hoặc lặp.
        \item \textbf{Loại bỏ có chủ đích:} \texttt{building\_id}, \texttt{timestamp}, nhãn, và nhóm \texttt{gte\_*} vì nguồn gốc và ánh xạ chưa được xác minh an toàn. \texttt{site\_id} được giữ làm biến phân loại.
    \end{itemize}

    \item \textbf{Giai đoạn 2: Huấn luyện rời rạc từng Mô hình (Baseline Modeling)}
    \begin{itemize}
        \item \textbf{Đã triển khai:} huấn luyện độc lập năm mô hình trên từng biến thể dữ liệu, theo cùng bộ fold: MLP (theo điểm), TCN nhân quả, BiGRU, BiLSTM (gán nhãn chuỗi có giám sát) và LSTM Autoencoder (tái tạo, chỉ học từ giờ bình thường).
        \item \textbf{Đã triển khai:} mỗi fold huấn luyện với AdamW, dừng sớm sau $5$ epoch không cải thiện, và \textbf{chọn checkpoint theo Average Precision (AP) trên fold kiểm định}. Điểm ngoài mẫu (\textit{out-of-fold}, OOF) được ghép từ $5$ fold, đồng thời mỗi checkpoint chấm luôn $T_{\text{out}}$.
        \item \textbf{Đã triển khai:} báo cáo AP, ROC-AUC, F1 tại ngưỡng chọn trên OOF, cùng AP và ROC-AUC trung bình theo toà nhà, cho cả OOF và $T_{\text{out}}$.
    \end{itemize}

    \item \textbf{Giai đoạn 3: Tối ưu hoá từng Mô hình}
    \begin{itemize}
        \item \textbf{Đã triển khai:} xử lý mất cân bằng bằng \texttt{BCEWithLogitsLoss} có trọng số lớp dương $w_{+} = 20$, cắt gradient ở $1{,}0$, điều chuẩn bằng dropout $0{,}2$ và weight decay $10^{-4}$, chọn checkpoint theo AP kiểm định.
        \item \textbf{Đã triển khai:} ngưỡng F1 được chọn trên điểm OOF của $T_{\text{in}}$ ($200$ phân vị) rồi \textbf{áp nguyên cho $T_{\text{out}}$}, nên F1 trên $T_{\text{out}}$ là ước lượng không dùng nhãn của tập kiểm tra để chọn ngưỡng.
        \item \textbf{Chưa triển khai (hướng mở rộng):} tìm siêu tham số bằng Optuna, so sánh BCE có trọng số với Focal Loss, quét $w_{+}$, chạy nhiều seed.
    \end{itemize}

    \item \textbf{Giai đoạn 4: Dung hợp Mô hình (Model Ensemble)}
    \begin{itemize}
        \item \textbf{Đã triển khai:} điểm trên $T_{\text{out}}$ của mỗi mô hình là trung bình điểm của $5$ checkpoint fold. Dung hợp giữa các mô hình bằng trung bình hạng phần trăm (\textit{percentile-rank}) với trọng số bằng nhau, phù hợp cả với xác suất của bộ phân loại lẫn sai số tái tạo có thang đo khác. Phép tính này được thực hiện bằng một đoạn mã hậu xử lý trên các tệp \texttt{oof\_<model>.parquet} và \texttt{holdout\_<model>.parquet}; lệnh \texttt{chunked\_pipeline} hiện chưa có bước dung hợp.
        \item \textbf{Chưa triển khai (hướng mở rộng):} trọng số học được và dung hợp với các mô hình Boosting của phần Học máy.
    \end{itemize}
\end{enumerate}

\subsection{Huấn luyện và Đánh giá Mô hình}

\subsubsection{Cơ sở Phương pháp luận: Mục tiêu Đầu ra, Tiêu chí Lựa chọn và Quy trình Thực hiện}

\paragraph{1. Mục đích Đầu ra của Bài toán (Output Objective):}\hspace{1cm}

Tại mỗi giờ $i$ của một toà nhà, mô hình trả về một điểm bất thường liên tục $s_i \in \mathbb{R}$. Với bộ phân loại có giám sát (MLP, TCN, BiGRU, BiLSTM), đó là xác suất:
\begin{equation}
    s_i = \hat{p}_i = \sigma(z_i) = P(y_i = 1 \mid \text{ngữ cảnh}_i) \in [0, 1]
\end{equation}
với $z_i$ là logit đầu ra. Với LSTM Autoencoder, đó là sai số tái tạo bình phương trung bình trên các kênh đầu vào:
\begin{equation}
    s_i = \frac{1}{d} \sum_{j=1}^{d} \left( \hat{x}_{i,j} - x_{i,j} \right)^2
\end{equation}
trong đó $d$ là số kênh đầu vào của biến thể đang xét ($28$, $31$ hoặc $35$). Điểm liên tục cho phép đánh giá không phụ thuộc ngưỡng bằng AP và ROC-AUC, và cho phép dung hợp mềm bằng hạng phần trăm. Khi cần nhãn nhị phân, quyết định là $\hat{y}_i = \mathbb{1}[s_i \geq \tau]$.

Hệ thống đánh giá theo ba độ đo:
\begin{itemize}
    \item \textbf{PR-AUC} (\textit{average precision}, AP; cùng định nghĩa \texttt{average\_precision\_score} của \texttt{scikit-learn} như phần Học máy), độ đo chính vì nhạy với lớp hiếm. Một mô hình đoán ngẫu nhiên có AP xấp xỉ tỷ lệ bất thường: $2{,}03\%$ trên $T_{\text{in}}$ và $2{,}45\%$ trên $T_{\text{out}}$.
    \item \textbf{ROC-AUC}, độ đo xếp hạng không phụ thuộc ngưỡng và tỷ lệ lớp.
    \item \textbf{F1-Score} tại ngưỡng cực đại hoá F1 trên $200$ phân vị của điểm trong OOF:
    \begin{equation}
        \tau^{*} = \arg\max_{\tau} \; \frac{2\,\text{Precision}(\tau)\,\text{Recall}(\tau)}{\text{Precision}(\tau) + \text{Recall}(\tau)}, \qquad F_1^{\max} = F_1^{\text{OOF}}(\tau^{*})
    \end{equation}
    Trên OOF, $F_1^{\max}$ là \textbf{cận trên mang tính chẩn đoán} vì ngưỡng được chọn ngay trên dữ liệu đang chấm điểm. Trên $T_{\text{out}}$, nhóm báo cáo $F_1(\tau^{*})$ với đúng $\tau^{*}$ lấy từ OOF, không điều chỉnh theo nhãn của $T_{\text{out}}$, nên đây là ước lượng F1 gần với triển khai thực tế.
\end{itemize}
Ngoài các độ đo gộp, nhóm tính AP và ROC-AUC \textbf{theo từng toà nhà} rồi lấy trung bình (macro) trên các toà nhà có cả hai lớp ($198$ toà ở OOF, $189$ toà ở $T_{\text{out}}$), để thấy mô hình có hoạt động đều giữa các toà nhà hay chỉ tốt ở một vài toà.

\paragraph{2. Tiêu chí Lựa chọn Thuật toán:}\hspace{1cm}

\begin{itemize}
    \item \textbf{Tiêu chí 1 - Khai thác được ngữ cảnh thời gian.} Nhiều bất thường chỉ nhận ra khi đặt giá trị đo trong mức tải thông thường của toà nhà theo giờ, thứ trong tuần và các giờ lân cận; ngữ cảnh $252$ giờ (hơn một tuần) được cấp cho các mô hình chuỗi để chúng so sánh với chu kỳ tuần.
    \item \textbf{Tiêu chí 2 - Tận dụng nhãn nhưng chịu được mất cân bằng.} Nhãn bất thường chỉ chiếm khoảng $2$--$2{,}5\%$; hàm mất mát có trọng số lớp được dùng thay cho SMOTE, vì nội suy giữa hai đoạn bất thường tạo ra chuỗi không có thật.
    \item \textbf{Tiêu chí 3 - Đa dạng hệ hình.} Năm mô hình đại diện bốn hệ hình khác nhau: theo điểm (MLP), tích chập nhân quả (TCN), hồi quy hai chiều (BiGRU, BiLSTM) và tái tạo không giám sát (LSTM-AE). Sự đa dạng này vừa phục vụ so sánh, vừa là điều kiện để dung hợp có ý nghĩa.
    \item \textbf{Tiêu chí 4 - Khả thi về tính toán.} Mỗi lần chạy đầy đủ gồm $3$ biến thể $\times$ $5$ mô hình $\times$ $5$ fold, trên khoảng $1{,}15$ triệu dòng huấn luyện; mô hình phải huấn luyện được trên GPU phổ thông (nhóm dùng batch $16$ cho các mô hình chuỗi và batch $1024$ cho MLP).
\end{itemize}

\paragraph{3. Các Bước Thực hiện Tuần tự trong Luồng Học Sâu (Execution Workflow):}\hspace{1cm}

Toàn bộ thực nghiệm chạy từ thư mục gốc của repo:
\begin{itemize}
    \item \textbf{Bước 1 (Gắn thời gian):} \texttt{python -m model\_DL.prepare\_chunked\_data} tạo sáu tệp parquet có \texttt{timestamp} và \texttt{chunked\_timestamp\_metadata.json}. Tệp đầu ra không bao giờ bị ghi đè.
    \item \textbf{Bước 2 (Chạy thử nhanh):} \texttt{python -m model\_DL.chunked\_pipeline train --smoke} (chọn $8$ toà nhà, $2$ fold, $1$ epoch) để kiểm tra toàn bộ đường ống trước khi chạy dài.
    \item \textbf{Bước 3 (Huấn luyện và đánh giá đầy đủ):} \texttt{python -m model\_DL.chunked\_pipeline train} chạy $3$ biến thể $\times$ $5$ mô hình $\times$ $5$ fold. Với mỗi mô hình và mỗi fold: khớp \texttt{FoldTransformer} trên các toà nhà huấn luyện, huấn luyện, chọn checkpoint theo AP, chấm OOF trên các toà nhà kiểm định và đồng thời chấm $T_{\text{out}}$.
    \item \textbf{Bước 4 (Tiếp tục khi bị gián đoạn):} thêm \texttt{--run-id <id> --resume}; các fold đã hoàn tất được dùng lại từ bộ nhớ đệm (\texttt{fold\_<n>\_<model>.npz}) sau khi kiểm tra độ phủ và giá trị hữu hạn. Mọi tệp (checkpoint, lịch sử epoch, dự đoán, bảng tổng kết) được ghi nguyên tử (ghi vào tệp tạm rồi đổi tên) và hệ điều hành bị chặn tự ngủ trong lúc chạy.
    \item \textbf{Bước 5 (Tính lại độ đo):} \texttt{python -m model\_DL.chunked\_pipeline evaluate --run-id <id>} tính lại độ đo từ các tệp OOF và $T_{\text{out}}$ đã lưu.
\end{itemize}
Các sản phẩm của mỗi lần chạy nằm trong \texttt{model\_DL/artifacts/chunked/<run-id>/}: \texttt{run\_config.json}, \texttt{summary.csv}, và với mỗi biến thể: \texttt{variant\_config.json}, \texttt{checkpoints/<model>/fold\_<n>.pt}, \texttt{logs/history\_<model>\_fold\_<n>.csv}, \texttt{logs/metrics\_<model>.json}, \texttt{predictions/oof\_<model>.parquet} và \texttt{predictions/holdout\_<model>.parquet}. Các bảng kết quả bên dưới được điền từ \texttt{summary.csv}, \texttt{metrics\_<model>.json} và hai tệp dự đoán nói trên của lần chạy \texttt{full\_local\_20261028}.

\paragraph{2. Giao thức đánh giá \texttt{GroupKFold} theo toà nhà trên $T_{\text{in}}$:}\hspace{1cm}

\begin{itemize}
    \item $200$ toà nhà của $T_{\text{in}}$ ($1{.}147{.}089$ dòng) được chia thành $5$ nhóm không giao nhau; mỗi lượt dùng $4$ nhóm để huấn luyện và nhóm còn lại để kiểm định. Mỗi toà nhà xuất hiện làm kiểm định đúng một lần, nên ghép $5$ lượt cho OOF phủ toàn bộ dòng của $T_{\text{in}}$.
    \item Bộ phân chia của \texttt{GroupKFold} là tất định theo \texttt{building\_id}, nên các mô hình khác (kể cả mô hình Học máy) có thể tái tạo đúng cùng các fold.
    \item Mã dựng chuỗi kiểm tra và báo lỗi nếu một lượt chia cắt ngang một toà nhà, để loại trừ rò rỉ do cửa sổ chồng lấp giữa huấn luyện và kiểm định; mỗi fold bắt buộc có cả hai lớp ở cả tập huấn luyện lẫn kiểm định.
    \item $T_{\text{out}}$ ($372.993$ dòng) được chấm bằng trung bình điểm của $5$ checkpoint fold; $T_{\text{out}}$ không tham gia huấn luyện, chọn checkpoint hay chọn ngưỡng.
\end{itemize}

\paragraph{Vùng nhìn của TCN.} Với $k = 3$ và hai tích chập giãn nở mỗi khối, vùng nhìn là $R = 1 + 2(k-1)\sum_{\ell} d_\ell = 1 + 2 \cdot 2 \cdot 63 = 253$ giờ. Giá trị này khớp với $252$ giờ ngữ cảnh cộng giờ hiện tại, nên giờ đích đầu tiên của mỗi đoạn nhìn được trọn vẹn phần ngữ cảnh (khi đoạn liên tục đủ dài).

\paragraph{Huấn luyện.} Tối ưu AdamW với tốc độ học $10^{-3}$ (không dùng bộ lập lịch), weight decay $10^{-4}$, batch $1024$ cho MLP và $16$ cho các mô hình chuỗi, dừng sớm sau $5$ epoch không cải thiện AP kiểm định, chạy với độ chính xác hỗn hợp tự động khi có GPU, hạt giống ngẫu nhiên $42 + \text{chỉ số fold}$. Số epoch tối đa của lần chạy chính thức là $10$; lần chạy được khởi động với trần $30$ epoch rồi hạ xuống $10$ khi tiếp tục (\texttt{--resume}), và các fold đã hoàn tất trước đó giữ nguyên checkpoint của chúng. Vì vậy một số fold của MLP trên biến thể Baseline chọn epoch tốt nhất ở các epoch $13$--$20$ (lớn hơn $10$). Hàm mất mát của các mô hình có giám sát là:
\begin{equation}
    \mathcal{L} = \frac{\sum_i m_i \cdot \left[ -w_{+}\, y_i \log \hat{p}_i - (1 - y_i)\log (1 - \hat{p}_i) \right]}{\sum_i m_i}, \qquad w_{+} = 20
\end{equation}
với $m_i$ là mặt nạ của các giờ đích hợp lệ (ở các đoạn đích ngắn, phần đệm bị loại). LSTM-AE tối thiểu hoá MSE tái tạo \textbf{chỉ trên các giờ đích bình thường} ($y_i = 0$) và dùng MSE của mọi giờ đích làm điểm bất thường. MLP không dùng ngữ cảnh chuỗi nên chỉ nhìn đặc trưng của đúng giờ đó (kể cả các đặc trưng ngữ cảnh $z$ của biến thể Context).

\subsubsection{Luận giải việc Lựa chọn và Loại trừ các Thuật toán}

\textbf{1. Nhóm được hiện thực và vai trò của từng mô hình}
\begin{itemize}
    \item \textbf{MLP (đối chứng):} cho biết các đặc trưng đã sinh sẵn (nhất là các đặc trưng ngữ cảnh $z$) giải quyết được bao nhiêu bài toán \textit{mà không cần mô hình chuỗi}. Nếu MLP ngang các mô hình chuỗi, lợi thế của kiến trúc chuỗi trên dữ liệu này nhỏ.
    \item \textbf{TCN nhân quả:} song song hoá được, huấn luyện nhanh, vùng nhìn tính trước được ($253$ giờ phủ chu kỳ tuần) và chỉ dùng quá khứ, nên đại diện cho kịch bản phát hiện gần thời gian thực.
    \item \textbf{BiGRU / BiLSTM:} dùng ngữ cảnh hai chiều trong cửa sổ $420$ giờ, nên biết cả những giờ sau giờ đang xét trong cùng cửa sổ. Đổi lại chỉ dùng được ngoại tuyến. GRU ít tham số hơn LSTM; cả hai được giữ để so sánh.
    \item \textbf{LSTM Autoencoder:} đại diện hệ hình tái tạo không giám sát phổ biến trong các nghiên cứu phát hiện bất thường chuỗi thời gian, dùng làm mốc so sánh với hướng có giám sát.
\end{itemize}

\textbf{2. Nhóm chưa hiện thực}
\begin{itemize}
    \item \textbf{GAN-LSTM / TAnoGAN:} có kết quả công bố trên LEAD nên có giá trị đối chiếu, nhưng mỗi cửa sổ cần khoảng $300$ bước tối ưu vector tiềm ẩn khi suy luận. Chấm điểm từng giờ trên toàn bộ dữ liệu cần cỡ $10^8$ lượt chạy bộ sinh, không khả thi trong khuôn khổ đề tài. Chỉ có thể tái hiện ở mức cửa sổ trên một tập con toà nhà.
    \item \textbf{USAD:} nhanh và ổn định hơn GAN, nhưng vẫn là hệ hình không giám sát và chưa có công bố trên dữ liệu năng lượng.
    \item \textbf{Anomaly Transformer, TranAD:} thiết kế cho chuỗi đa biến không giám sát với quy tắc đánh giá điều chỉnh theo đoạn; chỉ có $200$ chuỗi nên rủi ro quá khớp cao.
    \item \textbf{Conv-AE, VAE:} cùng hệ hình tái tạo với LSTM-AE; thêm vào tăng chi phí mà không mở góc nhìn mới.
    \item \textbf{RNN thuần:} triệt tiêu gradient trên chuỗi dài hàng trăm bước; LSTM/GRU thay thế hoàn toàn.
\end{itemize}

\textbf{Tiểu kết:} lựa chọn dựa trên ba đặc tính của bài toán: dữ liệu có nhãn nhưng mất cân bằng nặng, bất thường gồm cả loại có giá trị đặc biệt lẫn loại phụ thuộc ngữ cảnh, và chỉ có $200$ chuỗi toà nhà. Các nhận định về ưu thế của từng hệ hình được xem là giả thuyết, kiểm chứng ở bảng thí nghiệm bên dưới.

Về AP, BiGRU hơn ở $3/5$ fold (fold 1: TCN hơn $0{,}010$; fold 4: TCN hơn $0{,}032$), và xếp hạng đảo chiều giữa OOF (TCN $0{,}689$ so với BiGRU $0{,}677$) và $T_{\text{out}}$ (BiGRU $0{,}754$ so với TCN $0{,}668$), nên không kết luận được mô hình nào tốt hơn theo AP chỉ từ OOF.
    \item \textbf{BiGRU hơn BiLSTM:} cao hơn ở $5/5$ fold theo ROC-AUC (hơn $0{,}007$ đến $0{,}025$) và $4/5$ fold theo AP (fold 4 thua $0{,}002$); trên $T_{\text{out}}$ hai mô hình sít sao (AP $0{,}754$ so với $0{,}740$).
    \item \textbf{LSTM-AE kém xa} các mô hình có giám sát ở mọi biến thể (AP trên $T_{\text{out}}$ từ $0{,}034$ đến $0{,}111$; ROC-AUC OOF trên Baseline và Zone gần $0{,}5$, tức gần như ngẫu nhiên). Kết quả phù hợp với H2; việc giải thích nguyên nhân cần phân tích recall theo loại bất thường (H3).
    \item \textbf{Dao động giữa các fold lớn hơn nhiều khác biệt giữa mô hình ở AP.} Ví dụ AP của BiGRU dao động từ $0{,}611$ (fold 1) đến $0{,}775$ (fold 0). Vì vậy ranking theo một con số gộp cần được đọc cùng bảng theo fold. Kết quả theo fold của hai biến thể Baseline và Zone nằm trong \texttt{metrics\_<model>.json} tương ứng, với độ biến thiên tương tự.
    \item \textbf{Tác dụng của đặc trưng ngữ cảnh.} Chuyển từ Baseline sang Context nâng AP của BiGRU trên $T_{\text{out}}$ từ $0{,}515$ lên $0{,}754$ và ROC-AUC từ $0{,}870$ lên $0{,}965$, trong khi biến thể Zone gần như không đổi so với Baseline (AP $0{,}515$ và $0{,}515$). Mức tăng này có phần do đặc trưng $z$ thông tin hơn trên $T_{\text{out}}$ so với trên $T_{\text{in}}$ (xem mục giới hạn).
\end{itemize}

\subsection{Giới hạn của Hiện thực và Hướng Mở rộng}

Khi đối chiếu mã nguồn với thiết kế, nhóm ghi nhận các giới hạn sau để diễn giải kết quả cho đúng mức:
\begin{enumerate}
    \item \textbf{Chọn checkpoint trên fold đang chấm (OOF):} checkpoint tốt nhất của mỗi fold được chọn theo AP trên chính fold kiểm định dùng để báo cáo OOF, tạo thiên lệch lạc quan nhỏ ở các cột OOF. Các cột $T_{\text{out}}$ không chịu thiên lệch này vì $T_{\text{out}}$ không tham gia chọn checkpoint.
    \item \textbf{$F_1^{\max}$ trên OOF là cận trên:} ngưỡng được chọn trên chính OOF; $F_1(\tau^{*})$ trên $T_{\text{out}}$ dùng ngưỡng đó và là ước lượng gần với thực tế, nhưng ngưỡng chọn trên $T_{\text{in}}$ có thể lệch khi tỷ lệ bất thường của $T_{\text{out}}$ khác ($2{,}45\%$ so với $2{,}03\%$).
    \item \textbf{OOF gộp từ nhiều checkpoint:} điểm của các fold đến từ $5$ checkpoint khác nhau, nên thang điểm có thể lệch giữa fold, nhất là với LSTM-AE. AP và ROC-AUC gộp có thể khác trung bình theo fold; bảng nên báo cả hai.
    \item \textbf{Chuẩn hoá nhóm trong TCN:} lớp \texttt{GroupNorm} chuẩn hoá theo toàn bộ trục thời gian của đoạn, nên về nguyên tắc thông tin của giờ sau có thể ảnh hưởng đến giờ trước trong cùng cửa sổ. Phần tích chập vẫn nhân quả, nhưng khẳng định ``chỉ dùng quá khứ'' cần kiểm tra bằng thí nghiệm đổi giá trị tương lai.
    \item \textbf{LSTM-AE không có điểm thắt (bottleneck):} bộ mã hoá xuất chuỗi $64$ chiều ở mọi bước và bộ giải mã nhận chuỗi này, nên mô hình có thể học gần như phép đồng nhất. Đầu vào chứa cả \texttt{meter\_reading} nên mô hình có thể tái tạo tốt cả giá trị bất thường. Cần xem xét vector tiềm ẩn nhỏ hơn hoặc nhiễu đầu vào.
    \item \textbf{Một seed duy nhất và trần epoch không đồng nhất:} cấu hình dùng seed $42$ (cộng chỉ số fold). Báo cáo chênh lệch nhỏ cần chạy tối thiểu $3$ seed. Ngoài ra lần chạy được khởi động với trần $30$ epoch rồi hạ xuống $10$ khi tiếp tục, nên một số fold của MLP trên Baseline huấn luyện lâu hơn các fold còn lại.
    \item \textbf{Phân phối đặc trưng ngữ cảnh $z$ khác nhau giữa $T_{\text{in}}$ và $T_{\text{out}}$:} AP của riêng \texttt{z\_meter\_local} là $0{,}209$ trên $T_{\text{in}}$ nhưng $0{,}370$ trên $T_{\text{out}}$ (tương tự với \texttt{z\_meter\_workday}: $0{,}228$ và $0{,}364$), nên AP của Học sâu trên Context cao hơn ở $T_{\text{out}}$ so với OOF (BiGRU: $0{,}754$ so với $0{,}677$). Nhóm cho rằng nguyên nhân là các thống kê theo toà nhà--giờ được khớp trên $T_{\text{in}}$ bao gồm cả chính các dòng đó và các điểm bất thường, làm điểm $z$ của bất thường bị co lại trên $T_{\text{in}}$ nhưng không bị co trên $T_{\text{out}}$; đây là giả thuyết, chưa được kiểm chứng riêng. Tỷ lệ bất thường cao hơn của $T_{\text{out}}$ cũng góp phần nâng AP. Ảnh hưởng này áp dụng cho cả Học máy.
    \item \textbf{Ensemble chưa tích hợp vào pipeline:} dung hợp được tính bằng đoạn mã hậu xử lý trên tệp dự đoán; chưa có lệnh trong \texttt{chunked\_pipeline}, chưa có trọng số học được, và chưa dung hợp với mô hình Boosting.
    \item \textbf{Chưa có Optuna, Focal Loss và giao thức mức cửa sổ:} đây là hướng mở rộng. Các thí nghiệm theo trọng số lớp (Bảng~\ref{tab:dl_pos_weight}) và theo loại bất thường (Bảng~\ref{tab:dl_recall_by_type}) cũng chưa chạy.
\end{enumerate}
