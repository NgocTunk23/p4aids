\subsection{Các phương pháp Học máy hiện có để giải quyết bài toán}
%Ghi rõ và ưu nhược điểm
Do dữ liệu đã có gán nhãn cho biến \texttt{anomaly}( $0$ biểu thị trạng thái bình thường, $1$ biểu thị trạng thái bất thường), họ thuật toán dựa trên cây quyết định (Tree-based Models) và phương pháp học tập hợp (Ensemble Learning) thường thể hiện hiệu suất vượt trội.

\subsubsection{XGBoost (eXtreme Gradient Boosting)}\hspace{1cm}

XGBoost~\cite{xgboost} tối ưu hóa hàm mục tiêu bằng phương pháp khai triển Taylor bậc hai, cho phép hội tụ nhanh hơn. Cấu trúc cây được phát triển theo từng tầng. Mô hình cuối cùng là tổng hợp tuần tự của $K$ cây quyết định:
\begin{equation*}
    \hat{y}_i = \sum_{k=1}^{K} f_k(\mathbf{x}_i), \quad f_k \in \mathcal{F}
\end{equation*}
trong đó $\mathcal{F}$ là không gian của các cây hồi quy CART. Tại mỗi bước lặp $t$, hàm mục tiêu được tối ưu bằng khai triển Taylor bậc hai:
\begin{equation*}
    \mathcal{L}^{(t)} \simeq \sum_{i=1}^{n} \left[ g_i f_t(\mathbf{x}_i) + \frac{1}{2} h_i f_t^2(\mathbf{x}_i) \right] + \Omega(f_t)
\end{equation*}
với $g_i = \partial_{\hat{y}^{(t-1)}} \ell(y_i, \hat{y}^{(t-1)})$ là gradient bậc nhất, $h_i = \partial^2_{\hat{y}^{(t-1)}} \ell(y_i, \hat{y}^{(t-1)})$ là gradient bậc hai của hàm mất mát và $\Omega(f_t)$ là hàm phạt cấu trúc cây:
\begin{equation*}
    \Omega(f) = \gamma T + \frac{1}{2} \lambda \sum_{j=1}^{T} w_j^2 + \alpha \sum_{j=1}^{T} |w_j|
\end{equation*}
trong đó $T$ là số lá, $w_j$ là trọng số tại lá $j$, $\gamma$ phạt số lượng lá, $\lambda$ là hệ số điều chuẩn $L_2$ và $\alpha$ là hệ số điều chuẩn $L_1$.\\

\textbf{Ưu điểm:}
\begin{itemize}
    \item \textbf{Hạn chế Overfitting:} Tích hợp các thành phần phạt $L_1$ ($\alpha$) và $L_2$ ($\lambda$) trực tiếp vào hàm mục tiêu $\Omega(f)$, giúp hạn chế hiệu quả hiện tượng overfitting khi mô hình tiếp xúc với dữ liệu nhiễu từ các cảm biến đo lường.
    \item \textbf{Xử lý mất cân bằng lớp:} Hỗ trợ điều chỉnh trọng số thông qua siêu tham số \texttt{scale\_pos\_weight}, gán penalty cao hơn cho trường hợp phân loại sai lớp thiểu số.
    \item \textbf{Xử lý giá trị khuyết:} Thuật toán tự học hướng mặc định tại mỗi nút chia cho các giá trị bị khuyết, phù hợp với dữ liệu cảm biến thực tế thường có nhiều giá trị bị thiếu.
\end{itemize}

\textbf{Nhược điểm:}
\begin{itemize}
    \item \textbf{Độ phức tạp thời gian:} Thuật toán phân nhánh tham lam chính xác đòi hỏi duyệt qua toàn bộ dữ liệu liên tục. Mặc dù đã có các cải tiến như Histogram-based, XGBoost vẫn yêu cầu thời gian tính toán khi xử lý tập dữ liệu có quy mô lớn.
    \item \textbf{Phát triển cây Level-wise:} Chiến lược phát triển theo tầng buộc tất cả các nút ở cùng một độ sâu phải được mở rộng, dẫn đến việc phân chia tại nhiều nút không cần thiết.
\end{itemize}

\subsubsection{LightGBM (Light Gradient Boosting Machine)}\hspace{1cm}

LightGBM~\cite{lightgbm} được thiết kế chuyên biệt để tối ưu hóa hiệu suất trên dữ liệu quy mô lớn thông qua hai kỹ thuật cốt lõi: GOSS (Gradient-based One-Side Sampling) và EFB (Exclusive Feature Bundling). Cây được phát triển theo cơ chế ưu tiên lá.

Kỹ thuật GOSS ước lượng độ lợi thông tin bằng cách giữ lại toàn bộ các mẫu có gradient lớn và lấy mẫu ngẫu nhiên từ các mẫu có gradient nhỏ, sau đó nhân trọng số bù:
\begin{equation*}
    \widetilde{V}_j(d) = \frac{1}{n} \left( \frac{\left(\sum_{x_i \in A_l} g_i + \frac{1-a}{b} \sum_{x_i \in B_l} g_i\right)^2}{n_l^j(d)} + \frac{\left(\sum_{x_i \in A_r} g_i + \frac{1-a}{b} \sum_{x_i \in B_r} g_i\right)^2}{n_r^j(d)} \right)
\end{equation*}
trong đó $A$ là tập các mẫu có gradient lớn, $B$ là tập mẫu ngẫu nhiên từ các mẫu có gradient nhỏ, $l$ và $r$ ký hiệu nhánh trái và phải tại ngưỡng $d$ của đặc trưng $j$.\\

Kỹ thuật EFB gộp các đặc trưng thưa có tính loại trừ lẫn nhau thành các gói, giảm số lượng đặc trưng từ $p$ xuống số lượng gói $p' \ll p$.\\

\textbf{Ưu điểm:}
\begin{itemize}
    \item \textbf{Hiệu suất tính toán ưu việt:} Kỹ thuật Histogram giúp giảm độ phức tạp thời gian tìm kiếm điểm chia nhánh, mang lại tốc độ huấn luyện nhanh.
    \item \textbf{Xử lý biến định danh trực tiếp:} Khai thác hiệu quả các đặc trưng phân loại xuất hiện nhiều giá trị khác nhau như \texttt{site\_id}, \texttt{primary\_use},... bằng cách phân chia tối ưu trên các danh mục mà không cần mã hóa One-Hot, giúp kiểm soát chiều dữ liệu.
    \item \textbf{Tiết kiệm bộ nhớ:} Kỹ thuật EFB giảm đáng kể số chiều hiệu dụng, trong khi biểu diễn Histogram chỉ lưu trữ các bin rời rạc thay vì toàn bộ giá trị liên tục, giúp tiết kiệm RAM khi xử lý dữ liệu hàng chục triệu dòng.
    \item \textbf{GOSS giữ độ chính xác khi lấy mẫu:} Thay vì lấy mẫu đồng đều, GOSS ưu tiên các mẫu có gradient lớn (đóng góp nhiều nhất vào việc học), nên giảm dữ liệu mà vẫn duy trì chất lượng mô hình.
\end{itemize}

\textbf{Nhược điểm:}
\begin{itemize}
    \item \textbf{Rủi ro quá khớp cục bộ:} Sự phát triển theo chiều sâu lá có thể dẫn đến việc mô hình học các mẫu nhiễu cục bộ. Cần giới hạn nghiêm ngặt các tham số \texttt{max\_depth} và \texttt{min\_data\_in\_leaf}.
    \item \textbf{Nhạy cảm với dữ liệu không đa dạng:} Chiến lược phát triển cây theo chiều sâu lá tạo ra các cây sâu không cân bằng, dễ overfitting trên các tập dữ liệu nhỏ. Tuy nhiên, hạn chế này ít ảnh hưởng trong bài toán hiện tại do quy mô dữ liệu rất lớn.
\end{itemize}

\subsubsection{CatBoost (Categorical Boosting)}\hspace{1cm}

Thuật toán CatBoost~\cite{catboost} xây dựng các cây quyết định đối xứng và sử dụng cơ chế Ordered Target Encoding. Đặc điểm cốt lõi của CatBoost là sử dụng hoán vị ngẫu nhiên $\sigma$ để tính giá trị mã hóa cho biến định danh, tránh rò rỉ thông tin mục tiêu:
\begin{equation*}
    \hat{x}_{\sigma_p}^i = \frac{\sum_{j=1}^{p-1} [x_{\sigma_j}^i = x_{\sigma_p}^i] \cdot y_{\sigma_j} + a \cdot P}{\sum_{j=1}^{p-1} [x_{\sigma_j}^i = x_{\sigma_p}^i] + a}
\end{equation*}
trong đó $\sigma_p$ là chỉ số của mẫu thứ $p$ trong hoán vị, $x^i$ là giá trị đặc trưng định danh thứ $i$, $y$ là nhãn mục tiêu, $P$ là giá trị trung bình toàn cục của nhãn và $a > 0$ là tham số Laplace smoothing.

Cây đối xứng sử dụng cùng một điều kiện chia tại tất cả các nút ở cùng một độ sâu, tạo ra cấu trúc cây cân bằng hoàn hảo với $2^d$ lá (với $d$ là chiều sâu). Việc dự đoán trở thành phép tra bảng $O(d)$.\\

\textbf{Ưu điểm:}
\begin{itemize}
    \item \textbf{Ngăn chặn rò rỉ dữ liệu:} Ordered Target Encoding tính toán giá trị trung bình mục tiêu dựa trên các mẫu quan sát trong quá khứ nhân tạo theo hoán vị $\sigma$, giúp mã hóa các biến định danh một cách an toàn và mạnh mẽ mà không gây rò rỉ thông tin nhãn.
    \item \textbf{Tốc độ suy luận nhanh:} Cấu trúc cây đối xứng cho phép mô hình được biểu diễn dưới dạng các bảng tra cứu với độ phức tạp suy luận $O(d)$, tăng tốc độ dự đoán lên nhiều lần so với các cây không đối xứng.
    \item \textbf{Giảm Bias:} Sử dụng nhiều hoán vị khác nhau trong quá trình huấn luyện, giảm Bias do rò rỉ thông tin trong Gradient Boosting truyền thống, đặc biệt hiệu quả trên dữ liệu có nhiều biến phân loại như \texttt{building\_id}, \texttt{primary\_use}.
    \item \textbf{Hạn chế Overfitting:} Cấu trúc cây đối xứng có ít tham số hơn cây tự do (chỉ $d$ điều kiện chia thay vì $2^d - 1$ nút nội), đóng vai trò như một dạng điều chuẩn ngầm.
\end{itemize}

\textbf{Nhược điểm:}
\begin{itemize}
    \item \textbf{Thời gian huấn luyện lớn:} Quy trình mã hóa Ordered Target Encoding và xây dựng cây đối xứng khiến CatBoost mất nhiều thời gian huấn luyện hơn so với LightGBM, đặc biệt khi bộ dữ liệu chứa lượng lớn các biến liên tục.
    \item \textbf{Khả năng biểu diễn hạn chế:} Ràng buộc cùng một điều kiện chia tại mỗi tầng có thể hạn chế khả năng biểu diễn các mẫu phức tạp, đòi hỏi cây phải sâu hơn để đạt cùng mức chính xác so với cây tự do.
\end{itemize}

\subsubsection{HistGradientBoosting (Histogram-based Gradient Boosting)}\hspace{1cm}

HistGradientBoosting~\cite{hist_gb} là thuật toán tăng cường độ dốc hiện đại được xây dựng trong thư viện Scikit-learn, lấy cảm hứng trực tiếp từ kỹ thuật Histogram-based của LightGBM và được nhóm Quán quân cuộc thi LEAD~\cite{lead_winner} lựa chọn làm một trong bốn trụ cột phân loại chính. Thuật toán được thiết kế chuyên biệt để giải quyết rào cản tốc độ và bộ nhớ khi huấn luyện trên các tập dữ liệu chuỗi thời gian quy mô lớn ($N > 10^6$ mẫu).

Khác biệt cốt lõi của HistGradientBoosting nằm ở khâu lượng tử hóa đặc trưng (Feature Quantization / Binning): Thay vì duyệt qua toàn bộ các giá trị thực liên tục và sắp xếp lại dữ liệu tại mỗi bước phân nhánh với độ phức tạp $O(N \log N)$, thuật toán ánh xạ trước các đặc trưng liên tục vào tối đa $K = 256$ thùng số nguyên 8-bit (\texttt{uint8}):
\begin{equation*}
    q(x_i) \in \{0, 1, \dots, K-1\}
\end{equation*}
Đặc biệt, thùng thứ $255$ được dành riêng để biểu diễn tự nhiên các giá trị bị khuyết (NaN), cho phép mô hình tự động phân nhánh tối ưu cho dữ liệu thiếu mà không cần phải thực hiện bước điền khuyết (Imputation) nhân tạo trước đó.

Tại mỗi nút của cây quyết định thứ $t$, thuật toán tích lũy trực tiếp Gradient bậc nhất $g_i$ và Gradient bậc hai Hessian $h_i$ vào các bin histogram:
\begin{equation*}
    G_k = \sum_{i: q(x_i) = k} g_i, \quad H_k = \sum_{i: q(x_i) = k} h_i, \quad k \in \{0, 1, \dots, K-1\}
\end{equation*}
Nhờ tính chất cộng dồn của Histogram, việc tìm kiếm điểm chia nhánh tối ưu $s^*$ chỉ tốn chi phí $O(K)$ độc lập với kích thước dữ liệu $N$. Hơn nữa, thuật toán áp dụng cơ chế trừ Histogram (\textit{Histogram Subtraction Trick}): Histogram của nút con bên phải được suy ra trực tiếp bằng phép trừ:
\begin{equation*}
    G_{\text{Right}} = G_{\text{Parent}} - G_{\text{Left}}, \quad H_{\text{Right}} = H_{\text{Parent}} - H_{\text{Left}}
\end{equation*}
giúp giảm một nửa khối lượng tính toán trên toàn bộ cây. Độ lợi phân chia được tối đa hóa theo hàm mất mát chính quy:
\begin{equation*}
    \Delta \mathcal{L} = \frac{1}{2} \left[ \frac{G_L^2}{H_L + \lambda} + \frac{G_R^2}{H_R + \lambda} - \frac{G^2}{H + \lambda} \right] - \gamma
\end{equation*}
trong đó $\lambda$ là hệ số điều chuẩn $L_2$ và $\gamma$ là ngưỡng phạt tạo nhánh mới.\\

\textbf{Ưu điểm:}
\begin{itemize}
    \item \textbf{Tốc độ huấn luyện và tối ưu bộ nhớ vượt trội:} Việc chuyển đổi toàn bộ mảng dữ liệu sang định dạng số nguyên \texttt{uint8} kết hợp song song hóa đa luồng OpenMP/Cython giúp HistGradientBoosting tiêu tốn ít bộ nhớ RAM và đạt tốc độ huấn luyện nhanh gấp hàng chục lần so với các thuật toán phân nhánh chính xác (Exact-split).
    \item \textbf{Tự động xử lý giá trị khuyết và biến phân loại:} Hỗ trợ trực tiếp giá trị thiếu (NaN) và biến định danh (\texttt{categorical\_features}) bằng cơ chế phân nhóm tối ưu dựa trên thứ tự gradient, hạn chế tối đa rủi ro rò rỉ dữ liệu khi tiền xử lý.
    \item \textbf{Tích hợp cơ chế Dừng sớm (Early Stopping) nội bộ:} Tự động trích xuất một phần dữ liệu nội bộ làm tập kiểm định để theo dõi điểm số đánh giá (ROC-AUC / Loss), tự động dừng huấn luyện khi mô hình bắt đầu có dấu hiệu bão hòa hoặc quá khớp.
\end{itemize}

\textbf{Nhược điểm:}
\begin{itemize}
    \item \textbf{Mất mát độ phân giải vi mô cục bộ:} Do dữ liệu liên tục bị ép vào $256$ thùng rời rạc, thuật toán có thể làm phẳng các biến động giá trị cực kỳ nhỏ (tuy nhiên với quy mô dữ liệu $1{,}75$ triệu dòng của LEAD, hạn chế này không gây ảnh hưởng tiêu cực).
    \item \textbf{Khả năng tùy biến mất cân bằng lớp chuyên sâu:} So với \texttt{scale\_pos\_weight} linh hoạt của XGBoost hoặc \texttt{class\_weights} của CatBoost, cơ chế phạt mất cân bằng nhị phân trong Scikit-learn đòi hỏi phải điều chỉnh gián tiếp thông qua trọng số mẫu (\texttt{sample\_weight}).
\end{itemize}
